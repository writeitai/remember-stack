1"""Replace P2 snapshots with PostgreSQL 19 live property graphs (D98). 

2 

3revision: p9_17_0038 

4 

5The property graphs are catalog metadata over private live relational views; 

6public result hydration remains on the evidence-rich ``memory_v1`` authority 

7views. The graph views copy no tenant rows and therefore need neither a 

8generation nor a publication pointer. Variable-length traversal remains in 

9bounded SQL helpers because PostgreSQL 19 deliberately ships fixed graph 

10patterns before quantified and shortest-path patterns. 

11 

12The development-only downgrade restores legacy graph objects but does not 

13widen ``memory_v1.document_crossrefs_live`` back to unresolved or tombstoned 

14targets. D98 is a clean cut with disposable prerelease storage; retaining the 

15stricter public privacy view is intentional even when schema objects downgrade. 

16"""

32CREATE SCHEMA IF NOT EXISTS rememberstack_graph_internal;

35CREATE VIEW rememberstack_graph_internal.entities_live AS 

36SELECT deployment.deployment_id, 

37 scoped.entity_id, scoped.canonical_name, scoped.profile_summary 

38FROM deployments AS deployment 

39CROSS JOIN LATERAL ( 

40 WITH RECURSIVE survivor_chain(entity_id, current_entity_id) AS ( 

41 SELECT source.entity_id, source.entity_id 

42 FROM entities AS source 

43 WHERE source.deployment_id = deployment.deployment_id 

44 UNION 

45 SELECT chain.entity_id, merged.merged_into 

46 FROM survivor_chain AS chain 

47 JOIN entities AS merged 

48 ON merged.deployment_id = deployment.deployment_id 

49 AND merged.entity_id = chain.current_entity_id 

50 WHERE merged.merged_into IS NOT NULL 

51 ), 

52 survivor_map AS MATERIALIZED ( 

53 SELECT chain.entity_id, chain.current_entity_id AS survivor_entity_id 

54 FROM survivor_chain AS chain 

55 JOIN entities AS terminal 

56 ON terminal.deployment_id = deployment.deployment_id 

57 AND terminal.entity_id = chain.current_entity_id 

58 AND terminal.merged_into IS NULL 

59 ), 

60 provenance AS MATERIALIZED ( 

61 SELECT survivor.survivor_entity_id AS entity_id 

62 FROM mentions AS mention 

63 JOIN chunks AS chunk 

64 ON chunk.deployment_id = mention.deployment_id 

65 AND chunk.chunk_id = mention.chunk_id 

66 JOIN document_versions AS version 

67 ON version.deployment_id = chunk.deployment_id 

68 AND version.version_id = chunk.version_id 

69 AND version.doc_id = mention.doc_id 

70 AND version.deleted_at IS NULL 

71 JOIN documents AS document 

72 ON document.deployment_id = version.deployment_id 

73 AND document.doc_id = version.doc_id 

74 AND document.deleted_at IS NULL 

75 CROSS JOIN LATERAL ( 

76 SELECT decision.entity_id 

77 FROM resolution_decisions AS decision 

78 WHERE decision.deployment_id = mention.deployment_id 

79 AND decision.mention_id = mention.mention_id 

80 AND decision.superseded_by IS NULL 

81 ORDER BY decision.decided_at DESC, decision.decision_id DESC 

82 LIMIT 1 

83 ) AS decided 

84 JOIN survivor_map AS survivor 

85 ON survivor.entity_id = decided.entity_id 

86 WHERE mention.deployment_id = deployment.deployment_id 

87 UNION ALL 

88 SELECT survivor.survivor_entity_id 

89 FROM documents AS document 

90 JOIN survivor_map AS survivor 

91 ON survivor.entity_id = document.document_entity_id 

92 WHERE document.deployment_id = deployment.deployment_id 

93 AND document.deleted_at IS NULL 

94 ) 

95 SELECT e.entity_id, e.canonical_name, e.profile_summary 

96 FROM entities AS e 

97 WHERE e.deployment_id = deployment.deployment_id 

98 AND e.status = 'active' 

99 AND EXISTS ( 

100 SELECT 1 

101 FROM provenance 

102 WHERE provenance.entity_id = e.entity_id 

103 ) 

104) AS scoped;

381_NEIGHBORHOOD_HELPER = r""" 

382CREATE FUNCTION memory_v1.graph_neighborhood( 

383 deployment_id uuid, 

384 start_entity_id uuid, 

385 max_depth integer DEFAULT 2, 

386 predicates text[] DEFAULT NULL, 

387 valid_at timestamptz DEFAULT NULL, 

388 believed_at timestamptz DEFAULT NULL, 

389 max_results integer DEFAULT 100, 

390 expansion_budget integer DEFAULT 2000, 

391 frontier_budget integer DEFAULT 1000, 

392 time_budget_ms integer DEFAULT 1000 

393) 

394RETURNS TABLE ( 

395 row_kind text, 

396 hops integer, 

397 relation_ids uuid[], 

398 node_ids uuid[], 

399 truncated boolean, 

400 truncation_reason text, 

401 examined_edges bigint, 

402 returned_paths bigint, 

403 effective_depth integer, 

404 effective_expansion_budget integer, 

405 effective_frontier_budget integer, 

406 effective_result_budget integer, 

407 effective_time_budget_ms integer, 

408 applied_valid_at timestamptz, 

409 applied_believed_at timestamptz 

410) 

411LANGUAGE plpgsql 

412STABLE 

413PARALLEL UNSAFE 

414SECURITY INVOKER 

415SET search_path = memory_v1, pg_catalog 

416AS $function$ 

417#variable_conflict use_column 

418DECLARE 

419 depth_cap integer := least(greatest(coalesce(max_depth, 2), 1), 4); 

420 expansion_cap integer := least(greatest(coalesce(expansion_budget, 2000), 1), 2000); 

421 frontier_cap integer := least(greatest(coalesce(frontier_budget, 1000), 1), 1000); 

422 result_cap integer := least(greatest(coalesce(max_results, 100), 1), 500); 

423 time_cap integer := least(greatest(coalesce(time_budget_ms, 1000), 1), 5000); 

424 started_at timestamptz := clock_timestamp(); 

425 clock_valid timestamptz := coalesce(valid_at, statement_timestamp()); 

426 clock_believed timestamptz := coalesce(believed_at, statement_timestamp()); 

427 frontier jsonb[] := ARRAY[ 

428 jsonb_build_object( 

429 'head', start_entity_id::text, 

430 'nodes', to_jsonb(ARRAY[start_entity_id]), 

431 'edges', '[]'::jsonb 

432 ) 

433 ]; 

434 next_frontier jsonb[] := ARRAY[]::jsonb[]; 

435 result_buffer jsonb[] := ARRAY[]::jsonb[]; 

436 seen_vertices uuid[] := ARRAY[start_entity_id]; 

437 current_path jsonb; 

438 candidate jsonb; 

439 path_nodes uuid[]; 

440 path_edges uuid[]; 

441 head_id uuid; 

442 neighbor_id uuid; 

443 edge_record record; 

444 unpacked_path_ordinal bigint; 

445 level integer; 

446 examined bigint := 0; 

447 was_truncated boolean := coalesce(max_depth, 2) > 4; 

448 reason text := CASE WHEN coalesce(max_depth, 2) > 4 THEN 'depth_budget' END; 

449BEGIN 

450 IF (valid_at IS NULL) <> (believed_at IS NULL) THEN 

451 RAISE EXCEPTION 'a bitemporal traversal takes both clocks or neither' 

452 USING ERRCODE = 'invalid_parameter_value'; 

453 END IF; 

454 

455 <<levels>> 

456 FOR level IN 1..depth_cap LOOP 

457 next_frontier := ARRAY[]::jsonb[]; 

458 unpacked_path_ordinal := NULL; 

459 -- Plan one indexed adjacency statement per BFS level, not one copy of the 

460 -- survivor/provenance views per frontier node. 

461 FOR edge_record IN 

462 SELECT path.current_path, path.path_ordinal, head.head_id, edge.* 

463 FROM unnest(frontier) WITH ORDINALITY 

464 AS path(current_path, path_ordinal) 

465 CROSS JOIN LATERAL ( 

466 SELECT (path.current_path ->> 'head')::uuid AS head_id 

467 ) AS head 

468 CROSS JOIN LATERAL ( 

469 SELECT candidate.* 

470 FROM ( 

471 SELECT h.relation_id, h.subject_entity_id, h.object_entity_id 

472 FROM rememberstack_graph_internal.relations_history AS h 

473 WHERE h.deployment_id = graph_neighborhood.deployment_id 

474 AND h.subject_entity_id = head.head_id 

475 AND ( 

476 (valid_at IS NULL 

477 AND h.ingested_at <= clock_believed 

478 AND h.invalidated_at IS NULL) 

479 OR 

480 (valid_at IS NOT NULL 

481 AND (h.ingested_at IS NULL OR h.ingested_at <= clock_believed) 

482 AND (h.invalidated_at IS NULL 

483 OR h.invalidated_at > clock_believed)) 

484 ) 

485 AND (h.valid_from IS NULL OR h.valid_from <= clock_valid) 

486 AND (h.valid_until IS NULL OR h.valid_until > clock_valid) 

487 AND (predicates IS NULL OR h.predicate = ANY(predicates)) 

488 UNION ALL 

489 SELECT h.relation_id, h.subject_entity_id, h.object_entity_id 

490 FROM rememberstack_graph_internal.relations_history AS h 

491 WHERE h.deployment_id = graph_neighborhood.deployment_id 

492 AND h.object_entity_id = head.head_id 

493 AND h.subject_entity_id <> head.head_id 

494 AND ( 

495 (valid_at IS NULL 

496 AND h.ingested_at <= clock_believed 

497 AND h.invalidated_at IS NULL) 

498 OR 

499 (valid_at IS NOT NULL 

500 AND (h.ingested_at IS NULL OR h.ingested_at <= clock_believed) 

501 AND (h.invalidated_at IS NULL 

502 OR h.invalidated_at > clock_believed)) 

503 ) 

504 AND (h.valid_from IS NULL OR h.valid_from <= clock_valid) 

505 AND (h.valid_until IS NULL OR h.valid_until > clock_valid) 

506 AND (predicates IS NULL OR h.predicate = ANY(predicates)) 

507 ) AS candidate 

508 ORDER BY candidate.relation_id 

509 ) AS edge 

510 ORDER BY path.path_ordinal, edge.relation_id 

511 LIMIT greatest(expansion_cap - examined + 1, 1) 

512 LOOP 

513 head_id := edge_record.head_id; 

514 IF unpacked_path_ordinal IS DISTINCT FROM edge_record.path_ordinal THEN 

515 current_path := edge_record.current_path; 

516 SELECT coalesce(array_agg(value::uuid), ARRAY[]::uuid[]) 

517 INTO path_nodes 

518 FROM jsonb_array_elements_text(current_path -> 'nodes') AS value; 

519 SELECT coalesce(array_agg(value::uuid), ARRAY[]::uuid[]) 

520 INTO path_edges 

521 FROM jsonb_array_elements_text(current_path -> 'edges') AS value; 

522 unpacked_path_ordinal := edge_record.path_ordinal; 

523 END IF; 

524 IF examined >= expansion_cap THEN 

525 was_truncated := true; 

526 reason := 'expansion_budget'; 

527 EXIT; 

528 END IF; 

529 examined := examined + 1; 

530 IF extract(epoch FROM clock_timestamp() - started_at) * 1000 > time_cap THEN 

531 was_truncated := true; 

532 reason := 'time_budget'; 

533 EXIT; 

534 END IF; 

535 neighbor_id := CASE 

536 WHEN edge_record.subject_entity_id = head_id 

537 THEN edge_record.object_entity_id 

538 ELSE edge_record.subject_entity_id 

539 END; 

540 IF neighbor_id = ANY(path_nodes) OR neighbor_id = ANY(seen_vertices) THEN 

541 CONTINUE; 

542 END IF;
