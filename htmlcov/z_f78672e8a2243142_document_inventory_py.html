1"""What a customer's memory actually holds, one document at a time. 

2 

3The counts a dashboard shows — how much arrived, how much failed — can be 

4answered from metering. *Which* document, under what name, ingested when, and 

5where it got to cannot: metering is deliberately content-free. Only the 

6deployment knows the names, so this read model lives here and is served from 

7the deployment's own hostname (D33). 

8 

9## Two things this must not get wrong 

10 

11**The newest version is not the current one.** A lineage's 

12`current_version_id` points at the snapshot that finished processing and is 

13being served. A document uploaded five minutes ago and still converting has no 

14current version at all; a document whose newest upload failed still points at 

15the older, working one. Listing by the current pointer hides precisely the 

16documents somebody opens this screen to find, so a row reports the newest 

17surviving version and carries a separate `serving` flag. 

18 

19**The order has to be immutable.** The obvious order is "most recently 

20touched", but that key moves: re-ingesting a document changes it, and 

21tombstoning a version changes it back. Paging against a moving key silently 

22drops and repeats rows — a document below the cursor that gains a version 

23jumps above it and is never returned, and one whose newest version is deleted 

24falls below a cursor it already passed and is returned twice. 

25 

26So the order is `first_seen_at`, stamped once when the lineage is created and 

27never updated. A newly ingested document still appears at the top, because its 

28lineage is new. A *re-ingested* one keeps its place, which is the truthful 

29answer for an inventory: it is the same document, first seen when it was first 

30seen. `ix_documents_inventory_order` covers exactly this order, so a page 

31costs the page size rather than a scan of the corpus. 

32""" 

33

48#: Most documents one page may carry. A caller asking for more gets this. 

49MAX_PAGE_SIZE = 200 

50 

51#: Default when the caller does not say. Sized for a first screen, not a scroll. 

52DEFAULT_PAGE_SIZE = 50 

53 

54_PAGE = text( 

55 """ 

56 SELECT d.doc_id, 

57 d.title, 

58 d.source_kind, 

59 d.source_uri, 

60 d.first_seen_at, 

61 v.version_id, 

62 v.version_no, 

63 v.status::text AS status, 

64 v.ingested_at, 

65 v.error, 

66 -- Read in the same statement as `latest`, so the two describe one 

67 -- snapshot. An earlier build bounded `latest` by a watermark and 

68 -- left this unbounded, which reported a document as failed and 

69 -- serving at once because a later version had become ready. 

70 EXISTS ( 

71 SELECT 1 FROM document_versions ready 

72 WHERE ready.deployment_id = d.deployment_id 

73 AND ready.doc_id = d.doc_id 

74 AND ready.status = 'ready' 

75 AND ready.deleted_at IS NULL 

76 ) AS serving 

77 FROM documents d 

78 JOIN LATERAL ( 

79 SELECT dv.version_id, dv.version_no, dv.status, dv.ingested_at, dv.error 

80 FROM document_versions dv 

81 WHERE dv.deployment_id = d.deployment_id 

82 AND dv.doc_id = d.doc_id 

83 -- Tombstoned versions are not the document's current state. Deleting 

84 -- a version sets `deleted_at` and leaves `status` alone, so a deleted 

85 -- snapshot still reads `ready` — reporting it as the newest version 

86 -- would show somebody a document they had asked to have removed, 

87 -- labelled as fine. 

88 AND dv.deleted_at IS NULL 

89 ORDER BY dv.version_no DESC 

90 LIMIT 1 

91 ) v ON TRUE 

92 WHERE d.deployment_id = :deployment_id 

93 AND d.deleted_at IS NULL 

94 -- Every parameter is cast. PostgreSQL cannot infer the type of a bare 

95 -- placeholder compared against NULL, and refuses the statement rather 

96 -- than guessing. 

97 AND (CAST(:status AS text) IS NULL OR v.status::text = CAST(:status AS text)) 

98 AND ( 

99 CAST(:cursor_at AS timestamptz) IS NULL 

100 OR (d.first_seen_at, d.doc_id) 

101 < (CAST(:cursor_at AS timestamptz), CAST(:cursor_doc AS uuid)) 

102 ) 

103 ORDER BY d.first_seen_at DESC, d.doc_id DESC 

104 LIMIT :limit 

105 """ 

106)

109class DocumentInventory: 

110 """Read the document lineages a deployment holds, newest lineage first.""" 

111 

112 def __init__(self, *, engine: Engine) -> None: 

113 """Bind the spine this inventory reads.""" 

114 self._engine = engine 

115 

116 def list_documents( 

117 self, 

118 *, 

119 deployment_id: UUID, 

120 limit: int = DEFAULT_PAGE_SIZE, 

121 cursor: str | None = None, 

122 status: DocumentStatusFilter | None = None, 

123 ) -> DocumentPage: 

124 """One page of documents, ordered by when each lineage was first seen. 

125 

126 Ordered by when the lineage was first seen, so a document ingested a 

127 moment ago is at the top — and one re-ingested a moment ago keeps its 

128 place, because it is the same document. The tie break on `doc_id` is 

129 what makes the cursor total: a bulk import gives many lineages one 

130 `first_seen_at`, and without it two of them can swap places between 

131 pages, dropping one and repeating the other. 

132 """ 

133 size = max(1, min(int(limit), MAX_PAGE_SIZE)) 

134 cursor_at, cursor_doc = _decode_cursor(cursor) 

135 with self._engine.connect() as connection: 

136 rows = ( 

137 connection.execute( 

138 _PAGE, 

139 { 

140 "deployment_id": deployment_id, 

141 "limit": size + 1, 

142 "status": status, 

143 "cursor_at": cursor_at, 

144 "cursor_doc": cursor_doc, 

145 }, 

146 ) 

147 .mappings() 

148 .all() 

149 ) 

150 # One extra row was asked for: its presence is how we know another page 

151 # exists without a second count query, which on a large corpus would 

152 # cost more than the page itself. 

153 more = len(rows) > size 

154 page = rows[:size] 

155 documents = tuple( 

156 DocumentSummary( 

157 doc_id=row["doc_id"], 

158 title=row["title"], 

159 source_kind=row["source_kind"], 

160 source_uri=row["source_uri"], 

161 first_seen_at=row["first_seen_at"], 

162 latest=DocumentVersionSummary( 

163 version_id=row["version_id"], 

164 version_no=row["version_no"], 

165 status=row["status"], 

166 ingested_at=row["ingested_at"], 

167 error=row["error"], 

168 ), 

169 serving=bool(row["serving"]), 

170 ) 

171 for row in page 

172 ) 

173 next_cursor = ( 

174 _encode_cursor(page[-1]["first_seen_at"], page[-1]["doc_id"]) 

175 if more and page 

176 else None 

177 ) 

178 return DocumentPage(documents=documents, cursor=next_cursor)

181def _encode_cursor(first_seen_at: datetime, doc_id: UUID) -> str: 

182 """Opaque keyset position: the sort key, never an offset. 

183 

184 An offset re-reads rows that shifted while somebody was paging. This 

185 carries the sort key itself, and because that key never changes, the 

186 position it names stays meaningful for as long as the reader takes. 

187 """ 

188 raw = f"{first_seen_at.isoformat()}|{doc_id}".encode() 

189 return base64.urlsafe_b64encode(raw).decode().rstrip("=") 

190

192def _decode_cursor(cursor: str | None) -> tuple[datetime | None, UUID | None]: 

193 """Read a cursor back, or refuse one that does not parse. 

194 

195 Rejecting beats silently restarting: handing page one to somebody who 

196 asked for page three repeats documents they have already seen and reads as 

197 duplicates in the corpus rather than as the bug it is. 

198 

199 This checks that a cursor *parses*, not that this deployment issued it — 

200 the values are a timestamp and a document id, both of which a caller could 

201 write. That is acceptable because the cursor is not a capability: the query 

202 is scoped to the caller's own deployment either way, so the worst a 

203 hand-written one can do is start somebody at an odd place in their own 

204 corpus. 

205 """ 

206 if cursor is None: 

207 return None, None 

208 padding = "=" * (-len(cursor) % 4) 

209 try: 

210 raw = base64.urlsafe_b64decode(cursor + padding).decode() 

211 at_text, doc_text = raw.split("|") 

212 return datetime.fromisoformat(at_text), UUID(doc_text) 

213 except (ValueError, UnicodeDecodeError) as error: 

214 raise ValueError("cursor is malformed") from error
