1"""The expansion budget for zip-based documents (OOXML, EPUB) (D138 §9). 

2 

3The 100 MB reading limit measures the compressed file; a zip can declare 

4far more once expanded. Before a zip package is parsed, its central 

5directory's declared sizes are checked against a budget; Python's zipfile 

6never decompresses a member past its declared size, so the budget bounds 

7what parsing can expand. 

8""" 

9 

10import io 

11from typing import Final 

12import zipfile 

13 

14from rememberstack.model import ConversionError 

15 

16MAX_UNCOMPRESSED_TOTAL_BYTES: Final = 500_000_000 

17MAX_UNCOMPRESSED_MEMBER_BYTES: Final = 200_000_000 

18"""Declared-size budget per package and per member (starting values).""" 

19 

20 

21def require_zip_within_budget(*, content: bytes, what: str) -> None: 

22 """Fail when ``content`` is not a zip or declares too much expanded data.""" 

23 try: 

24 with zipfile.ZipFile(io.BytesIO(content)) as archive: 

25 sizes = [info.file_size for info in archive.infolist()] 

26 except (zipfile.BadZipFile, OSError, ValueError) as err: 

27 raise ConversionError(f"the {what} is not a readable zip package") from err 

28 if max(sizes, default=0) > MAX_UNCOMPRESSED_MEMBER_BYTES: 

29 raise ConversionError( 

30 f"the {what} declares a member larger than the " 

31 f"{MAX_UNCOMPRESSED_MEMBER_BYTES:,}-byte expansion limit" 

32 ) 

33 if sum(sizes) > MAX_UNCOMPRESSED_TOTAL_BYTES: 

34 raise ConversionError( 

35 f"the {what} expands to more than the " 

36 f"{MAX_UNCOMPRESSED_TOTAL_BYTES:,}-byte expansion limit" 

37 )
