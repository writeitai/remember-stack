1"""Which cost class a conversion route bills, and how its quantity is counted. 

2 

3A managed deployment reports one measurement per new source version before 

4the work that calls model providers (D139 and the cloud's D94 meter). The 

5conversion route decides the class: 

6 

7- ``pdf`` — every PDF, every page through OCR — is ``doc-scan``: the page 

8 count from the PDF page tree, taken before any OCR call. 

9- A local reading route (text, Markdown, code, HTML, office documents, email, 

10 notebooks, spreadsheets, delimited files, datasets) is ``doc-text``: the 

11 whitespace-normalized character count of the ``document.md`` it produced. 

12 These routes call no model provider, so counting their output still 

13 precedes every paid stage. 

14- ``card`` reads nothing and is free: it sends no measurement. 

15 

16Any other route has no managed cost class, so a metered deployment refuses 

17to run it rather than process unbilled provider work. 

18""" 

19 

20from __future__ import annotations 

21 

22from typing import Final 

23 

24from rememberstack.model.metering import CostClass 

25 

26METERING_CLASSIFIER_VERSION: Final = "route-cost-class-v1" 

27"""Pins the route → cost-class table below.""" 

28 

29DOC_TEXT_ALGORITHM_VERSION: Final = "document-md-unicode-whitespace-scalars-v1" 

30"""Pins how ``doc-text`` is counted: :func:`normalized_character_count` of 

31the route's ``document.md``.""" 

32 

33DOC_SCAN_ALGORITHM_VERSION: Final = "pdf-page-tree-v1" 

34"""Pins how ``doc-scan`` is counted: the PDF's structural page count.""" 

35 

36FREE_ROUTES: Final = frozenset({"card"}) 

37"""Routes that read nothing; their storage is billed, their processing is not.""" 

38 

39_ROUTE_CLASSES: Final[dict[str, CostClass]] = { 

40 "pdf": "doc-scan", 

41 "passthrough": "doc-text", 

42 "text": "doc-text", 

43 "markitdown": "doc-text", 

44 "office": "doc-text", 

45 "email": "doc-text", 

46 "notebook": "doc-text", 

47 "spreadsheet": "doc-text", 

48 "table": "doc-text", 

49 "dataset": "doc-text", 

50} 

51 

52 

53class UnmeteredRouteError(ValueError): 

54 """A metered deployment was asked to run a route that has no cost class.""" 

55 

56 

57def cost_class_for_route(*, route: str) -> CostClass: 

58 """The cost class a non-free route bills; refuse a route with none.""" 

59 try: 

60 return _ROUTE_CLASSES[route] 

61 except KeyError: 

62 raise UnmeteredRouteError( 

63 f"route {route!r} has no managed cost class (rate_class_unavailable)" 

64 ) from None 

65 

66 

67def normalized_character_count(*, text: str) -> int: 

68 """Count Unicode scalars after collapsing each maximal whitespace run to one.""" 

69 count = 0 

70 in_whitespace = False 

71 for scalar in text: 

72 whitespace = scalar.isspace() 

73 if whitespace: 

74 if not in_whitespace: 

75 count += 1 

76 else: 

77 count += 1 

78 in_whitespace = whitespace 

79 return count
