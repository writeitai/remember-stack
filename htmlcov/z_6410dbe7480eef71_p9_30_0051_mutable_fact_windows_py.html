162CREATE OR REPLACE VIEW memory_v1.facts_visible_history ( 

163 deployment_id, -- The deployment that owns the fact. 

164 fact_kind, -- Which fact layer this row belongs to, either relation or observation.

169 statement, -- The canonical statement of an observation, null for a relation. 

170 fact_label, -- Human-readable sentence for the fact, null when no label has been generated. 

171 valid_from, -- Canonical inclusive world-time start; NULL means unknown. 

172 valid_until, -- Canonical exclusive world-time end; NULL means unknown unless valid_precision is open. 

173 ingested_at, -- Raw transaction-time start: when the system first believed the fact. 

174 invalidated_at, -- Raw transaction-time end: when the system learned the fact was superseded, null while it is still believed. 

175 contradiction_group, -- Shared identifier of an unadjudicated contradiction, null when the fact is in no contradiction group. 

176 confidence, -- Aggregate confidence over the fact's evidence, null when never scored. 

177 evidence_count_current, -- LIVE count of distinct current-testimony lineages supporting the fact, read now and never a historical reconstruction. 

178 contradict_count_current, -- LIVE count of distinct current-testimony lineages contradicting the fact, read now and never a historical reconstruction.

180 valid_precision -- Chosen world-date precision; unknown and partial boundaries are distinct from explicitly open. 

181) AS 

182SELECT 

183 fact.deployment_id, 

184 fact.fact_kind, 

185 fact.fact_id, 

186 fact.subject_entity_id, 

187 fact.predicate, 

188 fact.object_entity_id, 

189 fact.statement, 

190 fact.fact_label, 

191 fact.valid_from, 

192 fact.valid_until, 

193 fact.ingested_at, 

194 fact.invalidated_at, 

195 fact.contradiction_group, 

196 fact.confidence, 

197 counts.supports, 

198 counts.contradicts, 

199 CASE WHEN EXISTS ( 

200 SELECT 1 

201 FROM review_queue AS queue 

202 WHERE queue.deployment_id = fact.deployment_id 

203 AND queue.item_kind = 'support_withdrawn'

210 SELECT base.*, coalesce(relation.valid_precision, observation.valid_precision)::text AS valid_precision

217CROSS JOIN LATERAL ( 

218 SELECT

220 count(*) FILTER (WHERE lineage.stance = 'contradicts')::bigint AS contradicts 

221 FROM v_memory_evidence_lineage_live AS lineage 

222 WHERE lineage.deployment_id = fact.deployment_id 

223 AND lineage.fact_kind = fact.fact_kind 

224 AND lineage.fact_id = fact.fact_id

Completed windows remain believed history while invalidated_at is NULL. Evidence counts and support state are current testimony, not reconstructed historical counts.';
