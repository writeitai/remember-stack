21def render_consumption_skill( 

22 *, context: ConsumptionSkillContext 

23) -> RenderedConsumptionSkill: 

24 """Render one complete ``SKILL.md`` from typed deployment state.""" 

25 deployment = context.deployment 

26 sections = ( 

27 _header(), 

28 _two_layer_headline(), 

29 _three_choices(), 

30 _deployment(context=context), 

31 _bound_examples(), 

32 _honesty_warnings(), 

33 _assured_operations(operations=context.operations), 

34 _grains(), 

35 _testimony(), 

36 _time_and_media(), 

37 _envelope(), 

38 _mounts(mounts=context.mounts), 

39 _working_rules(), 

40 ) 

41 return RenderedConsumptionSkill( 

42 deployment_id=deployment.deployment_id, 

43 version=CONSUMPTION_SKILL_VERSION, 

44 content="\n\n".join(sections).rstrip() + "\n", 

45 )

83def _deployment(*, context: ConsumptionSkillContext) -> str: 

84 """Render deployment identity, language, scopes, and current K state.""" 

85 deployment = context.deployment 

86 description = ( 

87 f"\n- Purpose: {_literal(value=deployment.description)}" 

88 if deployment.description 

89 else "" 

90 ) 

91 scopes = _scope_lines(scopes=deployment.scopes) 

92 knowledge_state = ( 

93 "known empty: no K pages are registered" 

94 if deployment.knowledge_page_count == 0 

95 else f"{deployment.knowledge_page_count} K page(s) are registered" 

96 ) 

97 empty_k = ( 

98 "This deployment currently has no K pages. Use open SQL or" 

99 " assured operations rather than inventing a missing summary. When" 

100 " mounts exist, fall back to the P3 corpus tree for orientation." 

101 if deployment.knowledge_page_count == 0 

102 else "If K returns `known_empty`, fall back to P3 or open query; never" 

103 " invent the missing synthesis." 

104 ) 

105 return ( 

106 "## This deployment\n\n" 

107 f"- Name: {_literal(value=deployment.name)}\n" 

108 f"- Slug: `{deployment.slug}`\n" 

109 f"- Deployment id: `{deployment.deployment_id}`\n" 

110 f"- Default language: `{deployment.default_language}`" 

111 f"{description}\n" 

112 f"- Plane K state: {knowledge_state}.\n" 

113 f"- Special-purpose scopes:\n{scopes}\n\n" 

114 f"{empty_k}" 

115 )

118def _bound_examples() -> str: 

119 """The same worked-example set discovery publishes (design §6).""" 

120 from rememberstack.core.open_query_prose import bound_worked_examples 

121 

122 sections: list[str] = ["## Bound worked examples"] 

123 for example in bound_worked_examples(): 

124 title = str(example["title"]) 

125 purpose = str(example["purpose"]) 

126 language = str(example["language"]) 

127 body = str(example["body"]) 

128 sections.append(f"### {title}\n\n{purpose}\n\n```{language}\n{body}\n```") 

129 return "\n\n".join(sections)

138def _assured_operations(*, operations: tuple[ConsumptionOperation, ...]) -> str: 

139 """Name the complete four-operation assured surface.""" 

140 core = {"resolve_entity", "testimony_context", "fact_context", "answer_context"} 

141 enabled_core = [operation for operation in operations if operation.name in core] 

142 if enabled_core: 

143 core_rows = "\n".join( 

144 f"- `{operation.name}` — `{operation.output_grain or 'ContextBundle/v1'}` / " 

145 f"`{operation.answer_intent}`: {_one_line(value=operation.description)}" 

146 for operation in enabled_core 

147 ) 

148 else: 

149 core_rows = ( 

150 "- The four assured operations are `resolve_entity`," 

151 " `testimony_context`, `fact_context`, and `answer_context`." 

152 ) 

153 return ( 

154 "## Assured operations\n\n" 

155 "Exactly four platform intent operations ship; three return D49" 

156 " Envelopes and `answer_context` returns `ContextBundle/v1`:\n\n" 

157 f"{core_rows}\n\n" 

158 "All other shipped retrieval patterns are open SQL or non-tool " 

159 "`examples.*` saved queries; there is no compatibility operation catalog." 

160 )

163def _grains() -> str: 

164 """Teach the claim-to-fact-to-compiled terminology ladder.""" 

165 return ( 

166 "## Keep the grains separate\n\n" 

167 "- A **claim** is immutable testimony: what one source asserted. It is " 

168 "evidence grain and may be stale, superseded, or contradicted.\n" 

169 "- A **relation** links two entities; an **observation** records a value or " 

170 "statement about one entity. Together they are the **fact layer**: the " 

171 "system's adjudicated, validity-filtered holdings. Questions of the form " 

172 '"is this true now?" go here by default.\n' 

173 "- A **compiled K page** is pre-paid synthesis. It is compiled grain and " 

174 "must be read with its compile time, stale flag, and open-flag count.\n" 

175 "- A **core belief** is a stricter configured K tier, not a new source of " 

176 "truth.\n\n" 

177 "Never blend evidence and facts into one unlabeled answer. If a task asks " 

178 "both what someone said and what the system believes, use " 

179 "`answer_context` and keep its complete `testimony` and `facts` child " 

180 "envelopes separate." 

181 )

184def _testimony() -> str: 

185 """Teach current testimony, historical opt-in, and withdrawn support.""" 

186 return ( 

187 "## Testimony currency and shaky support\n\n" 

188 "Claim search and claim views default to **current testimony**. Claims " 

189 "left behind by a living document's newer version or by a newer " 

190 "extraction generation are history, not current search results. " 

191 "`claims_as_of` means **what sources asserted as of a past system time**; " 

192 "it never means what is true now. Use `examples.claims_as_of` through " 

193 "`run_saved_query`, or query `claims_visible_history` with an inclusive " 

194 "overlap predicate.\n\n" 

195 "A fact with `support: withdrawn` has lost all current-testimony support " 

196 "because a toolchain re-read did not re-derive it. It still stands while " 

197 "review is open, but it is shaky: report the caveat, inspect its transcript " 

198 "and evidence, and do not make it load-bearing without verification." 

199 )

202def _time_and_media() -> str: 

203 """Keep the three media/time coordinates and derivation labels distinct.""" 

204 return ( 

205 "## Time and media\n\n" 

206 'Do not collapse these into "the timestamp":\n\n' 

207 "- a source locator such as `start_ms` says **where in a file** evidence " 

208 "occurs;\n" 

209 "- `valid_from` / `valid_until` say **when a fact held in the world**;\n" 

210 "- `ingested_at` / `believed_at` say **when the system knew it**.\n\n" 

211 "Live graph traversal shares PostgreSQL authority and has no snapshot" 

212 " generation. Its bounded-work status still matters: inspect truncation" 

213 " before treating absence as exhaustive.\n\n" 

214 "For media-derived evidence, read `evidence_mode`: `source_expression` is " 

215 "rendered speech/text, `model_observation` is what a model reports seeing, " 

216 "and `model_interpretation` is the model's interpretation. When tone or a " 

217 "visual detail matters, follow the source locator to the raw interval or " 

218 "region. The transcript is a map, not the territory." 

219 )

222def _envelope() -> str: 

223 """Teach response honesty fields for both Envelope and QueryResult.""" 

224 return ( 

225 "## Read the whole response\n\n" 

226 "For assured operations, check Envelope `grain`, required " 

227 "`temporal_scope`, identity regime, per-store freshness, " 

228 "truncation/continuation, and `dropped_by_hydration`. A result inside a " 

229 "live contradiction group must include or point to its co-members; " 

230 "report the competing sides instead of silently picking one.\n\n" 

231 "For open SQL, every answer is `QueryResult/v1`. Inspect " 

232 "`grade`, `truncated`/`truncation_reason`, warnings, " 

233 "`graph_invocations` for per-helper work status, " 

234 "confirmation/nomination drop counts, and `saved_query` stamps. " 

235 "Empty SQL is untyped; never promote it to a D49 negative.\n\n" 

236 "Negative Envelope results require different moves:\n\n" 

237 "- `unknown_entity`: widen resolution or search;\n" 

238 "- `known_empty`: the entity exists but no matching result is known within " 

239 "the stated freshness;\n" 

240 "- `boundary`: re-plan using the named workaround.\n\n" 

241 "Hard-forgotten material is intentionally indistinguishable from content " 

242 "that never existed." 

243 )

246def _mounts(*, mounts: PublishedMounts | None) -> str: 

247 """Render exact mount paths or the unmounted parity rule.""" 

248 if mounts is None: 

249 availability = ( 

250 "No mounts are available in this harness. Use API, CLI, or MCP for " 

251 "orientation, readable artifacts, and query operations." 

252 ) 

253 else: 

254 availability = ( 

255 "The four read-only mounts are available:\n\n" 

256 f"- P3 corpus tree: `{mounts.p3}`\n" 

257 f"- E0 artifacts: `{mounts.artifacts}`\n" 

258 f"- raw originals (off the navigation path; audited): `{mounts.raw}`\n" 

259 f"- plane K checkout: `{mounts.knowledge}`" 

260 ) 

261 return ( 

262 "## Filesystem first when mounts exist\n\n" 

263 f"{availability}\n\n" 

264 "When mounts exist, prefer them for navigation, reading, and grep. Reserve " 

265 "API/CLI/MCP for operations with no filesystem equivalent: open SQL, " 

266 "semantic/lexical SRFs, graph traversal, temporal as-of queries, " 

267 "hydration, transcripts, and deltas. Start in P3 or K, not raw. Follow an " 

268 "explicit raw pointer only when the original is needed, and use the " 

269 "deployment's audited raw-access mechanism." 

270 )

273def _working_rules() -> str: 

274 """End with a compact operational checklist for the consuming agent.""" 

275 return ( 

276 "## Before acting on a memory answer\n\n" 

277 "1. Did I use facts, not claims, for a current-truth question?\n" 

278 "2. Did I keep fact, evidence, and compiled grains labeled separately?\n" 

279 "3. For graph traversal, did I inspect the terminal work-budget status?\n" 

280 "4. Did I inspect caps, drops, truncation, contradictions, and withdrawn " 

281 "support?\n" 

282 "5. Did I keep world-validity and system-belief clocks paired?\n" 

283 "6. Did I hydrate to evidence or raw source when the stakes required it?" 

284 )

287def _scope_lines(*, scopes: tuple[ConsumptionScope, ...]) -> str: 

288 """Render special-purpose scope rows without inventing a default K page.""" 

289 if not scopes: 

290 return " - none registered" 

291 return "\n".join( 

292 f" - `{scope.slug}` ({_one_line(value=scope.name)})" 

293 + (f" at `{scope.git_path}`" if scope.git_path else "") 

294 + (f": {_one_line(value=scope.description)}" if scope.description else "") 

295 for scope in scopes 

296 )

299def _literal(*, value: str) -> str: 

300 """Render deployment-controlled prose as one explicit JSON string literal.""" 

301 literal = json.dumps(value, ensure_ascii=False).replace("`", "'") 

302 return f"`{literal}`"

306 """Collapse deployment-controlled display text to one Markdown-safe line."""
