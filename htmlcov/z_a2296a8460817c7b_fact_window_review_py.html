1"""The ``review_fact_windows`` stage (D140 §8.2). 

2 

3Two kinds of job share the stage: 

4 

5- **A lineage job** (target ``document``, a fresh request id; the lineage is in 

6 the payload) is queued, in the transaction of the change, whenever a lineage 

7 that ever declared effective periods may have moved its in-force intervals: 

8 a period write or clear, a period declared on an existing version at ingest, 

9 the completion of one of its versions, a deletion. Its first page passes the 

10 deployment's period-fence barrier; it then walks the lineage's live facts in 

11 bounded pages, queues one fact job for each fact whose period basis differs 

12 from the one its window was adjudicated against, and continues in a new job 

13 when a page is full. It makes no model call. 

14- **A fact job** (target ``relation`` or ``observation``) reviews that fact's 

15 window (:class:`~rememberstack.spine.fact_window_review.FactWindowReviewer`). 

16 A fact whose basis already matches is a no-op, so duplicate jobs are free. 

17"""

106 def _fan_out(self, *, work: ClaimedWork) -> HandlerOutcome: 

107 """One page of the lineage: queue reviews for moved facts, then continue. 

108 

109 The first page passes the deployment's period-fence barrier, so every 

110 window decision validated on inputs older than the change that queued 

111 this job has committed before stored bases are read. Each page is one 

112 bounded transaction; a full page queues a continuation job with its 

113 cursor, so progress is durable and memory stays bounded. 

114 """ 

115 doc_id = _payload_uuid(work=work, field="doc_id") 

116 payload = work.payload or {} 

117 after_kind, after_id = payload.get("after_kind"), payload.get("after_id") 

118 after = ( 

119 (str(after_kind), str(after_id)) 

120 if isinstance(after_kind, str) and isinstance(after_id, str) 

121 else None 

122 ) 

123 with self._engine.begin() as connection: 

124 if after is None: 

125 period_fence_barrier_on( 

126 connection=connection, deployment_id=work.deployment_id 

127 ) 

128 moved, next_after = facts_needing_review_on( 

129 connection=connection, 

130 deployment_id=work.deployment_id, 

131 doc_id=doc_id, 

132 after=after, 

133 limit=self._page_size, 

134 )

136 EnqueueWork( 

137 deployment_id=work.deployment_id, 

138 target_kind=ProcessingTarget.RELATION 

139 if kind == "relation" 

140 else ProcessingTarget.OBSERVATION, 

141 # A work row is unique per target: one review per lineage 

142 # job and fact, so a later change (even back to an earlier 

143 # basis) reviews again, and a retried fan-out does not. 

144 target_id=uuid5( 

145 NAMESPACE_URL, f"fact-window-review:{work.processing_id}:{fact_id}"

147 stage=PipelineStage.REVIEW_FACT_WINDOWS,
