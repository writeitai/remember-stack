1"""Verify signed keys at the engine perimeter (D136 §7.2–§7.4). 

2 

3The reference signed-credential adapter for the D61 auth-perimeter port, 

4beside the self-host ``HashedBearerAuth``. It is operator-generic: any key 

5issuer that produces this claim contract works. 

6 

7## Why a signature rather than a shared secret 

8 

9``HashedBearerAuth`` compares a presented secret against **one** digest in 

10configuration: one credential per deployment. This adapter keeps the issuer's 

11**public keys** instead and checks a signature, so the number of credentials in 

12circulation stops being a property of the deployment's configuration, and a 

13compromised deployment can only verify credentials, never mint them. 

14 

15## What a credential must carry 

16 

17A bearer is ``<prefix>_<JWS>``, where the prefix is letters only (remember.dev 

18uses ``rmb_``) so secret scanners can recognise a leaked key. The JWS is 

19EdDSA-signed; its ``kid`` names a key in the fetched key set **and** in the 

20accepted revocation document's ``active_kids``. Common claims: ``iss`` equal 

21to the configured issuer, a single-string ``aud``, a non-empty ``jti`` not in 

22the revocation document, ``iat``/``nbf``/``exp``, ``permissions`` (an array of 

23strings) and ``kind``. Per kind: 

24 

25======== ======================== ===================================== ========== 

26kind ``aud`` ``projects`` ``sub`` 

27======== ======================== ===================================== ========== 

28key ``org:<tenant>`` ``"org:*"`` or 1–20 ids incl. this one a person 

29session this deployment's id exactly ``[this project]`` a person 

30service this deployment's id — ``dpcred:<jti>`` 

31======== ======================== ===================================== ========== 

32 

33``key`` and ``session`` also carry ``org`` equal to the configured tenant. 

34``session`` may carry ``src`` (which host derived it); it is recorded for 

35audit and never decides authority. Any other ``aud`` — for example an OAuth 

36token meant for a hosted MCP server — is refused. 

37 

38## Fail closed 

39 

40Nothing signed is accepted without a fresh accepted revocation document 

41(:mod:`rememberstack.adapters.managed.perimeter_trust`). Every refusal is the 

42same opaque "no" to the caller; the reason class is logged with the ``jti`` 

43once the signature has verified, never the token. 

44"""

67#: A compact JWS begins with a base64url JSON header, which always encodes to 

68#: ``eyJ`` (``{"``). 

69_JWS_START = "eyJ" 

70 

71_LETTERS_PREFIX = re.compile(r"[A-Za-z]+") 

72 

73#: The machine subject marker (D60): a ``service`` credential names itself. 

74_MACHINE_SUBJECT_PREFIX = "dpcred:" 

75 

76#: ``projects`` value covering every project of the tenant, including ones 

77#: created after the key was minted. 

78_ALL_PROJECTS = "org:*" 

79 

80_MAX_PROJECTS = 20 

81 

82_COMMON_CLAIMS = ( 

83 "iss", 

84 "aud", 

85 "sub", 

86 "permissions", 

87 "kind", 

88 "iat", 

89 "nbf", 

90 "exp", 

91 "jti", 

92)

94#: Memory permissions this build can bound, and the engine scope each grants. 

95_MEMORY_PERMISSIONS = { 

96 "memory:read": PerimeterScope.READ, 

97 "memory:write": PerimeterScope.WRITE, 

98 "memory:ingest": PerimeterScope.INGEST, 

99}

102class SignedTokenUnusable(Exception): 

103 """This adapter refuses the presented credential; the message is the reason class."""

106def strip_credential_prefix(*, presented: str) -> str: 

107 """Remove a letters-only ``<prefix>_`` in front of a JWS. 

108 

109 Only when the bearer does not already start with a JWS header; everything 

110 up to and including the first ``_`` goes. A prefix that is not letters only 

111 is refused rather than guessed at. 

112 """ 

113 if presented.startswith(_JWS_START): 

114 return presented 

115 prefix, separator, rest = presented.partition("_") 

116 if not separator or not _LETTERS_PREFIX.fullmatch(prefix): 

117 raise SignedTokenUnusable("malformed_prefix") 

118 return rest

121def scope_for_permissions(*, permissions: object) -> PerimeterScope | None: 

122 """Map a credential's permissions to one engine scope (§7.4). 

123 

124 Permissions without the ``memory:`` prefix belong to another service and are 

125 ignored. An unknown ``memory:`` permission is refused, never downgraded: a 

126 permission this build cannot bound is not treated as a narrower one. 

127 ``memory:write`` covers everything; otherwise exactly one of 

128 ``memory:read`` or ``memory:ingest`` may be present. ``None`` means no 

129 memory permission: the credential authenticates and may do nothing. 

130 """ 

131 if not isinstance(permissions, list) or not all( 

132 isinstance(item, str) for item in permissions 

133 ): 

134 raise SignedTokenUnusable("malformed_permissions") 

135 memory = {item for item in permissions if item.startswith("memory:")} 

136 if memory - _MEMORY_PERMISSIONS.keys(): 

137 raise SignedTokenUnusable("unknown_permission") 

138 if "memory:write" in memory: 

139 return PerimeterScope.WRITE 

140 if len(memory) > 1: 

141 raise SignedTokenUnusable("conflicting_permissions") 

142 if not memory: 

143 return None 

144 return _MEMORY_PERMISSIONS[memory.pop()]

148 """Authenticate a signed key against the issuer's current trust documents."""

150 def __init__( 

151 self, 

152 *, 

153 deployment_id: UUID, 

154 issuer: str, 

155 tenant_id: str, 

156 project_id: str, 

157 trust: PerimeterTrust, 

158 ) -> None: 

159 """Bind this deployment's identity at the issuer and its trust source.""" 

160 self._deployment_id = deployment_id 

161 self._issuer = issuer 

162 self._tenant_id = tenant_id 

163 self._project_id = project_id 

164 self._trust = trust

166 def authenticate(self, *, credential: PerimeterCredential) -> AuthenticatedContext: 

167 """Verify a signed credential, or raise ``SignedTokenUnusable``.""" 

168 if credential.scheme.lower() != "bearer": 168 ↛ 169line 168 didn't jump to line 169 because the condition on line 168 was never true

169 raise SignedTokenUnusable("unsupported_scheme") 

170 current = self._trust.current() 

171 if current is None: 

172 raise SignedTokenUnusable("stale_revocation") 

173 keys, document = current 

174 

175 presented = credential.value.get_secret_value().decode("utf-8", errors="strict") 

176 token = strip_credential_prefix(presented=presented) 

177 try: 

178 header = jwt.get_unverified_header(token) 

179 except jwt.PyJWTError as error: 

180 raise SignedTokenUnusable("unreadable_header") from error 

181 if header.get("typ") == REVOCATION_TYPE: 

182 raise SignedTokenUnusable("wrong_type") 

183 kid = header.get("kid") 

184 # Named, never searched: trying every key would let a retired key keep 

185 # working through rotation. 

186 if not isinstance(kid, str) or kid not in keys: 

187 raise SignedTokenUnusable("unknown_kid") 

188 if kid not in document.active_kids: 

189 raise SignedTokenUnusable("retired_kid") 

190 

191 try: 

192 claims = jwt.decode( 

193 token, 

194 key=keys[kid], # type: ignore[arg-type] 

195 algorithms=[ACCEPTED_ALGORITHM], 

196 issuer=self._issuer, 

197 leeway=CLOCK_LEEWAY_SECONDS, 

198 options={ 

199 "require": list(_COMMON_CLAIMS), 

200 "verify_signature": True, 

201 "verify_iss": True, 

202 "verify_exp": True, 

203 "verify_nbf": True, 

204 "verify_iat": True, 

205 # The audience depends on `kind`; checked below. 

206 "verify_aud": False, 

207 }, 

208 ) 

209 except jwt.PyJWTError as error: 

210 raise SignedTokenUnusable("invalid_token") from error 

211 

212 for name in ("iat", "nbf", "exp"): 

213 value = claims[name] 

214 # PyJWT accepts any number; the contract is an integer epoch second. 

215 if not isinstance(value, int) or isinstance(value, bool): 

216 raise SignedTokenUnusable("malformed_time") 

217 token_id = claims["jti"] 

218 if not isinstance(token_id, str) or not token_id: 

219 raise SignedTokenUnusable("missing_jti") 

220 try: 

221 context = self._context(claims=claims, token_id=token_id) 

222 if token_id in document.revoked: 

223 raise SignedTokenUnusable("revoked") 

224 except SignedTokenUnusable as error: 

225 logger.info( 

226 "signed credential refused", 

227 extra={"reason": str(error), "jti": token_id}, 

228 ) 

229 raise 

230 return context

232 def _context( 

233 self, *, claims: Mapping[str, Any], token_id: str 

234 ) -> AuthenticatedContext: 

235 """Apply the per-kind claim rules and narrow into the perimeter's vocabulary.""" 

236 audience = claims["aud"] 

237 subject = claims["sub"] 

238 if not isinstance(audience, str): 

239 raise SignedTokenUnusable("malformed_aud") 

240 if not isinstance(subject, str) or not subject: 

241 raise SignedTokenUnusable("malformed_sub") 

242 scope = scope_for_permissions(permissions=claims["permissions"]) 

243 

244 kind = claims["kind"] 

245 # The narrow upload permission exists only for short-lived session 

246 # credentials; a long-lived key or a machine credential carrying it is 

247 # not something an issuer mints. 

248 if kind != "session" and "memory:ingest" in claims["permissions"]: 

249 raise SignedTokenUnusable("ingest_outside_session") 

250 source: str | None = None 

251 if kind == "key": 

252 if audience != f"org:{self._tenant_id}": 

253 raise SignedTokenUnusable("wrong_aud") 

254 self._check_org(claims=claims) 

255 projects = claims.get("projects") 

256 if projects != _ALL_PROJECTS and not ( 

257 isinstance(projects, list) 

258 and 1 <= len(projects) <= _MAX_PROJECTS 

259 and all(isinstance(item, str) for item in projects) 

260 and self._project_id in projects 

261 ): 

262 raise SignedTokenUnusable("not_covered") 

263 credential_kind = CredentialKind.KEY 

264 person: str | None = subject 

265 elif kind == "session": 

266 if audience != str(self._deployment_id): 

267 raise SignedTokenUnusable("wrong_aud") 

268 self._check_org(claims=claims) 

269 if claims.get("projects") != [self._project_id]: 

270 raise SignedTokenUnusable("not_covered") 

271 if "src" in claims: 

272 source = claims["src"] 

273 if not isinstance(source, str) or not source: 

274 raise SignedTokenUnusable("malformed_src") 

275 credential_kind = CredentialKind.BROWSER 

276 person = subject 

277 elif kind == "service": 

278 if audience != str(self._deployment_id): 

279 raise SignedTokenUnusable("wrong_aud") 

280 # A machine credential names itself; anything else would let a 

281 # payload choose which actor the audit records. 

282 if subject != f"{_MACHINE_SUBJECT_PREFIX}{token_id}": 

283 raise SignedTokenUnusable("wrong_sub") 

284 credential_kind = CredentialKind.DEPLOYMENT 

285 person = None 

286 else: 

287 raise SignedTokenUnusable("unknown_kind") 

288 

289 return AuthenticatedContext( 

290 deployment_id=self._deployment_id, 

291 principal="signed-bearer", 

292 subject=person, 

293 credential_id=token_id, 

294 credential_kind=credential_kind, 

295 source=source, 

296 scope=scope, 

297 )

299 def _check_org(self, *, claims: Mapping[str, Any]) -> None: 

300 if claims.get("org") != self._tenant_id: 

301 raise SignedTokenUnusable("wrong_org")
