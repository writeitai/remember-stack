1"""Canonical half-open bounds for D41 claim windows (D107 §5, WP-T.0a). 

2 

3Two IMMUTABLE functions turn a stored claim window — inclusive ends plus a 

4precision — into the half-open interval its precision means, so that every 

5comparison in the engine (the as-of candidate scan, the D106 temporal rung) 

6reads the same bounds: a day is the whole calendar day, a year is the whole 

7year, an instant is a non-empty point, an open window has no end. The Python 

8twin is ``rememberstack.core.temporal.canonical_bounds``; both are pinned by 

9tests to stay equivalent. Claim storage and its CHECK constraints are 

10unchanged. 

11 

12revision: p9_26_0047 

13""" 

14 

15from alembic import op 

16 

17revision: str = "p9_26_0047" 

18down_revision: str | None = "p9_25_0046" 

19branch_labels = None 

20depends_on = None

22_FUNCTIONS_DDL = r""" 

23CREATE FUNCTION claim_canonical_start( 

24 valid_from timestamptz, 

25 valid_precision claim_valid_precision 

26) RETURNS timestamptz 

27LANGUAGE sql IMMUTABLE PARALLEL SAFE AS $$ 

28 -- All calendar arithmetic runs on a UTC-naive timestamp 

29 -- (`AT TIME ZONE 'UTC'` in, `AT TIME ZONE 'UTC'` out), because 

30 -- `timestamptz + interval` uses the SESSION time zone's calendar fields; 

31 -- the result must not depend on the session that computed it (IMMUTABLE, 

32 -- and it feeds an expression index). 

33 SELECT CASE 

34 WHEN valid_from IS NULL OR valid_precision = 'unknown' THEN NULL 

35 WHEN valid_precision = 'day' THEN date_trunc('day', valid_from AT TIME ZONE 'UTC') AT TIME ZONE 'UTC' 

36 WHEN valid_precision = 'month' THEN date_trunc('month', valid_from AT TIME ZONE 'UTC') AT TIME ZONE 'UTC' 

37 WHEN valid_precision = 'quarter' THEN date_trunc('quarter', valid_from AT TIME ZONE 'UTC') AT TIME ZONE 'UTC' 

38 WHEN valid_precision = 'year' THEN date_trunc('year', valid_from AT TIME ZONE 'UTC') AT TIME ZONE 'UTC' 

39 ELSE valid_from 

40 END 

41$$; 

42COMMENT ON FUNCTION claim_canonical_start(timestamptz, claim_valid_precision) IS 

43 'D107 §5: the inclusive start of a claim window aligned to its precision unit in UTC; NULL for unknown precision.'; 

44 

45CREATE FUNCTION claim_canonical_end( 

46 valid_from timestamptz, 

47 valid_until timestamptz, 

48 valid_precision claim_valid_precision 

49) RETURNS timestamptz 

50LANGUAGE sql IMMUTABLE PARALLEL SAFE AS $$ 

51 SELECT CASE 

52 WHEN valid_from IS NULL OR valid_precision = 'unknown' THEN NULL 

53 WHEN valid_precision = 'open' THEN NULL 

54 WHEN valid_precision = 'instant' THEN valid_from + interval '1 microsecond' 

55 WHEN valid_precision = 'day' THEN (date_trunc('day', coalesce(valid_until, valid_from) AT TIME ZONE 'UTC') + interval '1 day') AT TIME ZONE 'UTC' 

56 WHEN valid_precision = 'month' THEN (date_trunc('month', coalesce(valid_until, valid_from) AT TIME ZONE 'UTC') + interval '1 month') AT TIME ZONE 'UTC' 

57 WHEN valid_precision = 'quarter' THEN (date_trunc('quarter', coalesce(valid_until, valid_from) AT TIME ZONE 'UTC') + interval '3 months') AT TIME ZONE 'UTC' 

58 WHEN valid_precision = 'year' THEN (date_trunc('year', coalesce(valid_until, valid_from) AT TIME ZONE 'UTC') + interval '1 year') AT TIME ZONE 'UTC' 

59 END 

60$$; 

61COMMENT ON FUNCTION claim_canonical_end(timestamptz, timestamptz, claim_valid_precision) IS 

62 'D107 §5: the EXCLUSIVE end of a claim window aligned to its precision unit in UTC; NULL for open or unknown.'; 

63 

64-- The as-of candidate scan filters on the canonical ends; an expression 

65-- index keeps it indexed exactly as the raw-column partial index did. 

66CREATE INDEX ix_claims_canonical_window 

67 ON claims ( 

68 deployment_id, 

69 claim_canonical_start(claim_valid_from, claim_valid_precision), 

70 claim_canonical_end(claim_valid_from, claim_valid_until, claim_valid_precision) 

71 ) 

72 WHERE claim_valid_precision <> 'unknown'; 

73"""
