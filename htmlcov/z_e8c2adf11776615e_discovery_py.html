120def search_query_space(*, query: str, k: int = 10) -> tuple[ViewDescription, ...]: 

121 """Rank manifest text (names, comments, grains) against a free-text query. 

122 

123 Deterministic and content-free: a simple term-overlap score over the 

124 checked-in manifest only (§6 binds discovery to manifest text). 

125 """ 

126 if not 1 <= k <= 25: 

127 raise SandboxRejection( 

128 code=QueryErrorCode.INVALID_PARAMETER, message="k must be in 1..25" 

129 ) 

130 terms = {term for term in query.lower().split() if term} 

131 if not terms: 131 ↛ 132line 131 didn't jump to line 132 because the condition on line 131 was never true

132 raise SandboxRejection( 

133 code=QueryErrorCode.INVALID_PARAMETER, message="query must be non-empty" 

134 ) 

135 scored: list[tuple[float, ViewDescription]] = [] 

136 for view in describe_query_space().views: 

137 name = view.name.lower() 

138 prose = " ".join((view.grain, view.comment, " ".join(view.row_key))).lower() 

139 # A term in the relation's own name is the strongest signal an agent 

140 # can give: "current facts" must find `facts_current`, not merely the 

141 # views whose prose mentions both words. 

142 score = sum(4.0 for term in terms if term in name) 

143 score += sum(1.0 for term in terms if term in prose) 

144 if name in terms: 144 ↛ 145line 144 didn't jump to line 145 because the condition on line 144 was never true

145 score += 8.0 

146 if score > 0: 

147 scored.append((score, view)) 

148 scored.sort(key=lambda pair: (-pair[0], pair[1].name)) 

149 return tuple(view for _, view in scored[:k])
