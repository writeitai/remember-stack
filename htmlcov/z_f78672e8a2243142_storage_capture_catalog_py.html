3These are native reference seeds, not classified storage members or proof of

68class NativeStoreReference(_NativeRecord): 

69 """One native store locator or journal intent, including unfinished state. 

70 

71 A tree/repository locator is a root to expand through qualified store 

72 evidence, never a fabricated object or a row-byte estimate. Only raw 

73 objects record a native length; missing length remains explicit. A journal 

74 intent carries the actual forget UUID, not an invented store path. Its 

75 qualified adapter derives the concrete journal key and proves presence. 

76 """ 

77 

78 native_class: str 

79 native_id: SecretStr = Field(repr=False)

105DECLARE 

106 native_scope uuid; 

107 native_identity jsonb; 

108 page jsonb; 

109BEGIN 

110 IF session_user <> __COLLECTOR__ THEN 

111 RAISE EXCEPTION USING ERRCODE='42501', MESSAGE='native_catalog_login'; 

112 END IF; 

113 IF current_setting('transaction_isolation') <> 'repeatable read' 

114 OR current_setting('transaction_read_only') <> 'on' THEN 

115 RAISE EXCEPTION USING ERRCODE='25001', MESSAGE='native_catalog_transaction'; 

116 END IF; 

117 IF page_limit IS NULL OR page_limit < 1 OR page_limit > 1024 

118 OR after_class IS NULL OR after_id IS NULL THEN 

119 RAISE EXCEPTION USING ERRCODE='22023', MESSAGE='native_catalog_page'; 

120 END IF; 

121 IF section = 'acquire' THEN 

122 IF after_class <> '' OR after_id <> '' THEN 

123 RAISE EXCEPTION USING ERRCODE='22023', MESSAGE='native_catalog_cursor'; 

124 END IF; 

125 LOCK TABLE public.deployments, public.alembic_version, 

126 public.content_objects, public.document_representations, 

127 public.document_structure_generations, public.document_reference_generations, 

128 public.projection_snapshots, public.knowledge_artifacts, 

129 public.knowledge_compilations, public.knowledge_plan_runs, 

130 public.forget_manifests, public.managed_ingest_measurements, 

131 public.managed_ingest_outcomes IN ACCESS SHARE MODE NOWAIT; 

132 ELSE 

133 IF (SELECT count(*) FROM public.deployments) <> 1 

134 OR (SELECT count(*) FROM public.alembic_version) <> 1 

135 OR (SELECT version_num FROM public.alembic_version) <> 'p9_46_0067' THEN 

136 RAISE EXCEPTION USING ERRCODE='55000', MESSAGE='native_catalog_scope'; 

137 END IF; 

138 SELECT deployment_id INTO STRICT native_scope FROM public.deployments; 

139 END IF; 

140 SELECT jsonb_build_object( 

141 'session_login', session_user, 

142 'database_name', current_database(), 

143 'database_oid', (SELECT oid::text FROM pg_database WHERE datname=current_database()), 

144 'postmaster_started_unix_us', 

145 floor(extract(epoch FROM pg_postmaster_start_time())*1000000)::text, 

146 'backend_pid', pg_backend_pid(), 

147 'virtual_transaction_id', 

148 (SELECT virtualtransaction FROM pg_locks WHERE pid=pg_backend_pid() 

149 AND locktype='virtualxid' AND mode='ExclusiveLock'), 

150 'snapshot_identity', pg_current_snapshot()::text) INTO native_identity; 

151 IF section = 'acquire' THEN 

152 RETURN jsonb_build_object('identity',native_identity,'deployment_id',NULL,'storage',NULL); 

153 ELSIF section = 'bind' THEN 

154 IF after_class <> '' OR after_id <> '' THEN 

155 RAISE EXCEPTION USING ERRCODE='22023', MESSAGE='native_catalog_cursor'; 

156 END IF; 

157 RETURN jsonb_build_object('identity',native_identity,'deployment_id',native_scope::text,'storage', 

158 (SELECT jsonb_build_object('raw_bucket',raw_bucket, 

159 'artifacts_bucket',artifacts_bucket,'corpusfs_bucket',corpusfs_bucket, 

160 'knowledge_repo_uri',knowledge_repo_uri) FROM public.deployments)); 

161 ELSIF section = 'identity' THEN 

162 IF after_class <> '' OR after_id <> '' THEN 

163 RAISE EXCEPTION USING ERRCODE='22023', MESSAGE='native_catalog_cursor'; 

164 END IF; 

165 page := '[]'::jsonb; 

166 ELSIF section = 'references' THEN 

167 WITH native_references AS ( 

168 SELECT 'content_objects.raw_uri'::text AS native_class, 

169 content_hash AS native_id,'raw'::text AS store_role,raw_uri AS locator, 

170 'object'::text AS locator_kind,byte_size::text AS recorded_bytes, 

171 NULL::text AS native_state,content_hash AS recorded_digest FROM public.content_objects 

172 WHERE deployment_id=native_scope 

173 UNION ALL 

174 SELECT 'document_representations.'||u.field,r.representation_id::text, 

175 'artifacts',u.locator,'object',NULL,r.status, 

176 CASE u.field WHEN 'conversion_uri' THEN r.manifest_hash 

177 WHEN 'markdown_uri' THEN r.markdown_hash 

178 WHEN 'pageindex_uri' THEN r.pageindex_hash ELSE NULL END 

179 FROM public.document_representations r 

180 CROSS JOIN LATERAL (VALUES ('markdown_uri',r.markdown_uri), 

181 ('pageindex_uri',r.pageindex_uri),('conversion_uri',r.conversion_uri), 

182 ('blocks_uri',r.blocks_uri),('meta_uri',r.meta_uri)) u(field,locator) 

183 WHERE r.deployment_id=native_scope AND u.locator IS NOT NULL 

184 UNION ALL 

185 SELECT 'document_structure_generations.pageindex_uri', 

186 structure_generation_id::text,'artifacts',pageindex_uri,'object',NULL,NULL,NULL 

187 FROM public.document_structure_generations 

188 WHERE deployment_id=native_scope AND pageindex_uri IS NOT NULL 

189 UNION ALL 

190 SELECT 'document_reference_generations.artifact_uri',generation_id::text, 

191 'artifacts',artifact_uri,'object',NULL,status,NULL 

192 FROM public.document_reference_generations 

193 WHERE deployment_id=native_scope AND artifact_uri IS NOT NULL 

194 UNION ALL 

195 SELECT CASE WHEN plane::text='P3_corpusfs' THEN 'projection_snapshots.gcs_uri' 

196 ELSE 'projection_snapshots.unsupported_origin' END, 

197 snapshot_id::text,'corpus',gcs_uri, 

198 'tree',NULL,status::text,NULL FROM public.projection_snapshots 

199 WHERE deployment_id=native_scope 

200 UNION ALL 

201 SELECT 'knowledge_artifacts.'||u.field,k.artifact_id::text,'knowledge',

203 FROM public.knowledge_artifacts k 

204 CROSS JOIN LATERAL (VALUES ('git_path',k.git_path), 

205 ('curation_path',k.curation_path)) u(field,locator)

207 UNION ALL

213 SELECT 'knowledge_plan_runs.session_transcript_uri',run_id::text, 

214 'artifacts',session_transcript_uri,'object',NULL,status,NULL 

215 FROM public.knowledge_plan_runs WHERE deployment_id=native_scope 

216 UNION ALL 

217 SELECT 'forget_manifests.portable_intent',forget_id::text,'forget', 

218 forget_id::text,'journal_intent',NULL,status::text,NULL 

219 FROM public.forget_manifests 

220 WHERE deployment_id=native_scope 

221 ), bounded AS ( 

222 SELECT * FROM native_references WHERE 

223 (native_class COLLATE "C",native_id COLLATE "C") > 

224 (after_class COLLATE "C",after_id COLLATE "C") 

225 ORDER BY native_class COLLATE "C",native_id COLLATE "C" LIMIT page_limit 

226 ) SELECT COALESCE(jsonb_agg(to_jsonb(bounded) ORDER BY

229 ELSIF section = 'terminals' THEN

231 RAISE EXCEPTION USING ERRCODE='22023', MESSAGE='native_catalog_cursor';

242 ELSE 

243 RAISE EXCEPTION USING ERRCODE='22023', MESSAGE='native_catalog_section'; 

244 END IF; 

245 RETURN jsonb_build_object('identity',native_identity, 

246 'deployment_id',native_scope::text,'rows',page); 

247END

251def install_native_capture_catalog( 

252 *, connection: psycopg.Connection, definer_role: str, collector_role: str 

253) -> None: 

254 """Install fixed reads using the guard's existing roles and private schema. 

255 

256 No role, membership, table SELECT grant or secret is created. This trusted 

257 operator operation is not called by a capture request or engine startup. 

258 """ 

259 if connection.info.transaction_status != TransactionStatus.IDLE: 

260 raise NativeRelationGuardRefused("native_catalog_installer_transaction") 

261 with connection.transaction(): 

262 boundary = connection.execute( 

263 "SELECT r.rolname=%s,has_function_privilege(%s,p.oid,'EXECUTE')," 

264 "n.nspowner=p.proowner FROM pg_catalog.pg_proc p " 

265 "JOIN pg_catalog.pg_roles r ON r.oid=p.proowner " 

266 "JOIN pg_catalog.pg_namespace n ON n.oid=p.pronamespace " 

267 "WHERE p.oid=to_regprocedure(" 

268 "'rememberstack_capture_internal.guard_relation_allocation(bigint)')", 

269 (definer_role, collector_role), 

270 ).fetchone() 

271 if boundary != (True, True, True) or definer_role == collector_role: 

272 raise NativeRelationGuardRefused("native_catalog_installer_boundary") 

273 body = _BODY.replace( 

274 "__COLLECTOR__", sql.Literal(collector_role).as_string(connection) 

275 ) 

276 function = sql.Identifier( 

277 "rememberstack_capture_internal", "read_capture_catalog" 

278 ) 

279 connection.execute( 

280 sql.SQL( 

281 "CREATE OR REPLACE FUNCTION {}(section text,after_class text," 

282 "after_id text,page_limit integer) RETURNS jsonb LANGUAGE plpgsql " 

283 "SECURITY DEFINER SET search_path=pg_catalog,pg_temp SET timezone='UTC' AS {}" 

284 ).format(function, sql.Literal(body)) 

285 ) 

286 connection.execute( 

287 sql.SQL("ALTER FUNCTION {}(text,text,text,integer) OWNER TO {}").format( 

288 function, sql.Identifier(definer_role) 

289 ) 

290 ) 

291 connection.execute( 

292 sql.SQL( 

293 "REVOKE ALL ON FUNCTION {}(text,text,text,integer) FROM PUBLIC" 

294 ).format(function) 

295 ) 

296 old_grantees = connection.execute( 

297 "SELECT DISTINCT r.rolname FROM pg_catalog.pg_proc p " 

298 "CROSS JOIN LATERAL pg_catalog.aclexplode(COALESCE(p.proacl," 

299 "pg_catalog.acldefault('f',p.proowner))) a " 

300 "JOIN pg_catalog.pg_roles r ON r.oid=a.grantee " 

301 "WHERE p.oid=to_regprocedure(%s) AND a.grantee<>p.proowner", 

302 (_SIGNATURE,), 

303 ).fetchall() 

304 for (old_grantee,) in old_grantees: 

305 connection.execute( 

306 sql.SQL( 

307 "REVOKE ALL ON FUNCTION {}(text,text,text,integer) FROM {}" 

308 ).format(function, sql.Identifier(old_grantee)) 

309 ) 

310 connection.execute( 

311 sql.SQL( 

312 "GRANT EXECUTE ON FUNCTION {}(text,text,text,integer) TO {}" 

313 ).format(function, sql.Identifier(collector_role)) 

314 )
