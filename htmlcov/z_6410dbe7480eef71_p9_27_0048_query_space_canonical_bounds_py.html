1"""Expose canonical bounds in the query space (D107 §5, WP-T.0b). 

2 

3``memory_v1.canonical_bounds`` wraps the public IMMUTABLE twins from 

4``p9_26_0047`` so open SQL can canonicalise a stored D41 window without 

5EXECUTE on the public schema. ``memory_v1.claims_canonical`` is 

6``claims_visible_history`` plus the half-open ``canon_start`` / ``canon_end`` 

7columns those twins compute. Claim storage is unchanged. 

8 

9revision: p9_27_0048 

10"""

27CANONICAL_BOUNDS_FUNCTION_DDL = r""" 

28CREATE FUNCTION memory_v1.canonical_bounds( 

29 valid_from timestamptz, 

30 valid_until timestamptz, 

31 valid_precision text 

32) 

33RETURNS TABLE(canon_start timestamptz, canon_end timestamptz) 

34LANGUAGE sql 

35IMMUTABLE 

36PARALLEL SAFE 

37SECURITY DEFINER 

38SET search_path = pg_catalog, public, pg_temp 

39ROWS 1 

40AS $$ 

41 SELECT 

42 claim_canonical_start( 

43 valid_from, valid_precision::claim_valid_precision 

44 ) AS canon_start, 

45 claim_canonical_end( 

46 valid_from, valid_until, valid_precision::claim_valid_precision 

47 ) AS canon_end 

48$$; 

49"""

51_CANONICAL_START_SQL = """ 

52CASE 

53 WHEN claim_valid_from IS NULL OR claim_valid_precision = 'unknown' THEN NULL 

54 WHEN claim_valid_precision = 'day' THEN date_trunc('day', claim_valid_from AT TIME ZONE 'UTC') AT TIME ZONE 'UTC' 

55 WHEN claim_valid_precision = 'month' THEN date_trunc('month', claim_valid_from AT TIME ZONE 'UTC') AT TIME ZONE 'UTC' 

56 WHEN claim_valid_precision = 'quarter' THEN date_trunc('quarter', claim_valid_from AT TIME ZONE 'UTC') AT TIME ZONE 'UTC' 

57 WHEN claim_valid_precision = 'year' THEN date_trunc('year', claim_valid_from AT TIME ZONE 'UTC') AT TIME ZONE 'UTC' 

58 ELSE claim_valid_from 

59 END 

60""".strip()

62CLAIMS_CANONICAL_VIEW_DDL = rf""" 

63CREATE VIEW memory_v1.claims_canonical ( 

64 deployment_id, -- The deployment that owns the claim. 

65 claim_id, -- Stable identity of this immutable claim. 

66 doc_id, -- The live lineage that asserted the claim. 

67 version_id, -- The non-tombstoned version the claim was extracted from. 

68 representation_id, -- The reading whose character offsets the claim's anchors use. 

69 chunk_id, -- The chunk the claim was extracted from. 

70 claim_text, -- The standalone assertion as extracted, which is source testimony rather than adjudicated truth. 

71 source_span, -- The verbatim slice of the source the claim derives from. 

72 char_start, -- Start character offset of source_span within the named representation's markdown. 

73 char_end, -- End character offset of source_span within the named representation's markdown. 

74 added_context, -- The substrings decontextualization added, each with the bundle source it came from. 

75 temporal_class, -- How the claim behaves over time, either static, dynamic, or atemporal; null when unclassified. 

76 is_attributed, -- True when the claim preserves an attribution, so it entails that someone said it rather than that it holds. 

77 audit_status, -- Result of the sampled independent grounding audit, defaulting to unaudited. 

78 kept_flagged, -- True when selection kept the claim but marked it for review. 

79 extractor_version, -- The extractor generation that produced the claim, which is part of the D54 extraction basis. 

80 asserted_at, -- Assertion-event time: when the source asserted this, null when the source carries no date. 

81 claim_valid_from, -- Immutable inclusive start of the world-time interval the SOURCE asserted, null for unbounded-before or unknown. 

82 claim_valid_until, -- Immutable inclusive end of that interval, null for open-per-source or unknown as disambiguated by claim_valid_precision. 

83 claim_valid_precision, -- Granularity of the asserted interval, from unknown through instant, day, month, quarter, and year to open. 

84 claim_valid_kind, -- Which world-interval was asserted, such as event_time or measurement_period; null when unclassified. 

85 ingested_at, -- Transaction-time: when this deployment extracted the claim. 

86 source_kind, -- The connector family of the asserting lineage. 

87 source_handle, -- Stable human-usable handle for the asserting lineage, formed from its connector-native identity. 

88 is_current_testimony, -- True while this claim is the current transcription of its chunk under D54; false once a newer extraction generation or a living-mode version move superseded it. 

89 canon_start, -- Inclusive start of the half-open canonical window (D107 §5); null when precision is unknown. 

90 canon_end -- Exclusive end of the half-open canonical window; null when the window is open or unknown. 

91) AS 

92SELECT 

93 c.deployment_id, 

94 c.claim_id, 

95 c.doc_id, 

96 ch.version_id, 

97 ch.representation_id, 

98 c.chunk_id, 

99 c.claim_text, 

100 c.source_span, 

101 c.char_start, 

102 c.char_end, 

103 c.added_context, 

104 c.temporal_class::text, 

105 c.is_attributed, 

106 c.audit_status::text, 

107 c.kept_flagged, 

108 c.extractor_version, 

109 c.asserted_at, 

110 c.claim_valid_from, 

111 c.claim_valid_until, 

112 c.claim_valid_precision::text, 

113 c.claim_valid_kind::text, 

114 c.ingested_at, 

115 dl.source_kind, 

116 dl.source_kind || ':' || coalesce(dl.source_ref, dl.doc_id::text), 

117 c.is_current_testimony, 

118 -- Inline the immutable twins' expressions: view ownership supplies table 

119 -- privileges, not EXECUTE on private functions. Keeping expressions visible 

120 -- also allows the planner to match ix_claims_canonical_window after inlining 

121 -- the index's SQL functions. Database tests pin all three projections equal. 

122 {_CANONICAL_START_SQL.replace("claim_valid_", "c.claim_valid_")}, 

123 CASE 

124 WHEN c.claim_valid_from IS NULL OR c.claim_valid_precision = 'unknown' THEN NULL 

125 WHEN c.claim_valid_precision = 'open' THEN NULL 

126 WHEN c.claim_valid_precision = 'instant' THEN c.claim_valid_from + interval '1 microsecond' 

127 WHEN c.claim_valid_precision = 'day' THEN (date_trunc('day', coalesce(c.claim_valid_until, c.claim_valid_from) AT TIME ZONE 'UTC') + interval '1 day') AT TIME ZONE 'UTC' 

128 WHEN c.claim_valid_precision = 'month' THEN (date_trunc('month', coalesce(c.claim_valid_until, c.claim_valid_from) AT TIME ZONE 'UTC') + interval '1 month') AT TIME ZONE 'UTC' 

129 WHEN c.claim_valid_precision = 'quarter' THEN (date_trunc('quarter', coalesce(c.claim_valid_until, c.claim_valid_from) AT TIME ZONE 'UTC') + interval '3 months') AT TIME ZONE 'UTC' 

130 WHEN c.claim_valid_precision = 'year' THEN (date_trunc('year', coalesce(c.claim_valid_until, c.claim_valid_from) AT TIME ZONE 'UTC') + interval '1 year') AT TIME ZONE 'UTC' 

131 END 

132-- Identical coordinate/visibility joins to claims_visible_history (0025). 

133-- Project directly from c so filtering does not require a second claims scan. 

134FROM public.claims AS c 

135JOIN public.chunks AS ch 

136 ON ch.deployment_id = c.deployment_id 

137 AND ch.chunk_id = c.chunk_id 

138 AND ch.doc_id = c.doc_id 

139JOIN memory_v1.document_versions_visible AS vv 

140 ON vv.deployment_id = ch.deployment_id 

141 AND vv.version_id = ch.version_id 

142 AND vv.doc_id = ch.doc_id 

143JOIN public.document_representations AS representation 

144 ON representation.deployment_id = ch.deployment_id 

145 AND representation.version_id = ch.version_id 

146 AND representation.representation_id = ch.representation_id 

147 AND representation.status = 'ready' 

148JOIN memory_v1.documents_live AS dl 

149 ON dl.deployment_id = c.deployment_id 

150 AND dl.doc_id = c.doc_id; 

151COMMENT ON VIEW memory_v1.claims_canonical IS 

152 'One row per historically visible claim with surviving lineage, keyed by (deployment_id, claim_id), carrying the stored inclusive D41 window beside the half-open canonical bounds that every overlap predicate must use (D107 §5). canon_start is inclusive and canon_end exclusive; both are null when precision is unknown, and canon_end is also null for an open window. Overlap is a.start < b.end AND b.start < a.end with a null end as unbounded. This relation is IMMUTABLE SOURCE TESTIMONY: it never answers what currently holds. Claims of forgotten lineages and tombstoned versions are absent.'; 

153"""

155CLAIMS_CLOCK_COMMENT_DDL = r""" 

156COMMENT ON VIEW memory_v1.claims_visible_history IS 

157 'One row per claim whose source lineage is live and whose source version is not tombstoned, keyed by (deployment_id, claim_id) and joined to documents_live on (deployment_id, doc_id) and to document_versions_visible on (deployment_id, version_id). This relation is IMMUTABLE SOURCE TESTIMONY and never answers what currently holds: claim_valid_from and claim_valid_until are the inclusive stored D41 window (an instant has equal endpoints). World-time overlap MUST use claims_canonical.canon_start / canon_end, which are the half-open canonical bounds; filtering these raw columns as a current-truth or overlap predicate is the wrong query — use facts_current for what holds now, and claims_canonical for as-of testimony. A null claim_valid_from means unbounded-before or unknown and a null claim_valid_until means open-per-source or unknown, disambiguated by claim_valid_precision. Claims of forgotten lineages and tombstoned versions are absent; is_current_testimony is D54 bookkeeping and never validity. The view carries no counts.'; 

158COMMENT ON VIEW memory_v1.claims_live IS 

159 'One row per current-testimony claim, keyed by (deployment_id, claim_id): the subset of claims_visible_history whose D54 currency flag is still set. Like every claim relation this is IMMUTABLE SOURCE TESTIMONY, and "live" here means current transcription of a live source, never current truth: a claim in this relation can be contradicted by the adjudicated worldview, and querying its validity window to answer what holds now is the wrong query — start from facts_current and follow fact_claim_evidence_live back to here. The stored claim_valid_* window is inclusive D41 storage; world-time overlap belongs on claims_canonical. Claims of forgotten lineages and tombstoned versions are absent, and this relation is the sole claim input to the D54 counting path. The view carries no counts.'; 

160"""
