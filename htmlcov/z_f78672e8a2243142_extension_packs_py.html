1"""Extension-pack installation (registries §4/§7, D15): enable a pack as a unit. 

2 

3Installing writes the pack's registry rows for one deployment — entity types, 

4predicates, and domain/range signatures, all `tier='extension'` with the pack 

5id — after verifying every anchor exists (extend-never-fork: an extension 

6type MUST declare a registered parent; a pack that forks is refused whole). 

7Idempotent: enabling twice is a no-op. 

8"""

27def install_pack(*, engine: Engine, deployment_id: UUID, pack: ExtensionPack) -> None: 

28 """Enable one pack for one deployment in a single transaction.""" 

29 with engine.begin() as connection: 

30 _require_anchors(connection=connection, deployment_id=deployment_id, pack=pack) 

31 _refuse_conflicts(connection=connection, deployment_id=deployment_id, pack=pack) 

32 connection.execute( 

33 _UPSERT_PACK, 

34 { 

35 "pack_id": pack.pack_id, 

36 "name": pack.name, 

37 "description": pack.description, 

38 }, 

39 ) 

40 connection.execute( 

41 _ENABLE_PACK, {"deployment_id": deployment_id, "pack_id": pack.pack_id} 

42 ) 

43 for entity_type in pack.entity_types: 

44 connection.execute( 

45 _INSERT_TYPE, 

46 { 

47 "deployment_id": deployment_id, 

48 "type": entity_type.type, 

49 "parent_type": entity_type.parent_type, 

50 "description": entity_type.description, 

51 "examples": list(entity_type.examples), 

52 "pack_id": pack.pack_id, 

53 }, 

54 ) 

55 for predicate in pack.predicates: 

56 connection.execute( 

57 _INSERT_PREDICATE, 

58 { 

59 "deployment_id": deployment_id,

79def _require_anchors( 

80 *, connection: Connection, deployment_id: UUID, pack: ExtensionPack 

81) -> None: 

82 """Refuse the whole pack if any anchor is not ALREADY registered (D15). 

83 

84 Extend-never-fork means every pack type declares a parent that exists in 

85 the deployment registry BEFORE the pack installs — a pack-local chain 

86 (Child anchored to another type from the same pack) is refused too 

87 (Codex review): the DDL's contract is a registered core-side parent. 

88 Signatures may reference pack types (they install together). 

89 """ 

90 pack_types = {entity_type.type for entity_type in pack.entity_types} 

91 registered = { 

92 row[0] 

93 for row in connection.execute(_SELECT_TYPES, {"deployment_id": deployment_id}) 

94 } 

95 for entity_type in pack.entity_types: 

96 if entity_type.parent_type not in registered: 

97 raise PackAnchorError( 

98 f"pack {pack.pack_id!r} type {entity_type.type!r} anchors to " 

99 f"{entity_type.parent_type!r}, which is not already registered " 

100 "(extend-never-fork, D15: pack-local anchors are refused)" 

101 ) 

102 signature_types = { 

103 entity_type 

104 for predicate in pack.predicates 

105 for pair in predicate.signatures 

106 for entity_type in pair 

107 } 

108 unknown = signature_types - (registered | pack_types) 

109 if unknown: 109 ↛ 110line 109 didn't jump to line 110 because the condition on line 109 was never true

110 raise PackAnchorError( 

111 f"pack {pack.pack_id!r} signatures reference unregistered " 

112 f"types {sorted(unknown)!r}" 

113 )

116def _refuse_conflicts( 

117 *, connection: Connection, deployment_id: UUID, pack: ExtensionPack 

118) -> None: 

119 """Unit installation (Codex review): an existing row under a pack name 

120 must BE this pack's row (same pack_id and parent) — anything else fails 

121 the whole install rather than silently blending registries.""" 

122 type_rows = { 

123 row[0]: (row[1], row[2]) 

124 for row in connection.execute( 

125 _SELECT_TYPE_DEFS, {"deployment_id": deployment_id} 

126 ) 

127 } 

128 for entity_type in pack.entity_types: 

129 existing = type_rows.get(entity_type.type) 

130 if existing is not None and existing != (entity_type.parent_type, pack.pack_id): 

131 raise PackConflictError( 

132 f"type {entity_type.type!r} already registered with " 

133 f"(parent, pack) = {existing!r}; pack {pack.pack_id!r} " 

134 "cannot install over it" 

135 ) 

136 predicate_rows = { 

137 row[0]: row[1] 

138 for row in connection.execute( 

139 _SELECT_PREDICATE_DEFS, {"deployment_id": deployment_id} 

140 ) 

141 } 

142 for predicate in pack.predicates: 

143 existing_pack = predicate_rows.get(predicate.predicate) 

144 if existing_pack is not None and existing_pack != pack.pack_id: 144 ↛ 145line 144 didn't jump to line 145 because the condition on line 144 was never true

145 raise PackConflictError( 

146 f"predicate {predicate.predicate!r} already registered by " 

147 f"pack {existing_pack!r}; pack {pack.pack_id!r} cannot " 

148 "install over it" 

149 )
