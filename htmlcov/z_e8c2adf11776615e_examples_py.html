1"""The shipped `examples.*` queries (§2, §5). 

2 

3These are starting points, not platform operations. The platform wrote them, so 

4they are honest about what they compute and every one parses through the same 

5grammar an ad-hoc statement does — but their MEANING is not a platform 

6guarantee. Copy one, change its filters, and the copy is customer-authored, 

7which is the intended use rather than a misuse.

426def example_operator_fixtures( 

427 name: str, *, handles: ExampleFixtureHandles 

428) -> dict[str, dict[str, object]]: 

429 """Materialize the four §5 operator fixtures for one shipped example. 

430 

431 Returns a mapping kind -> {parameters, max_rows?} suitable for building 

432 `OperatorFixture` values at the call site (keeps this module free of the 

433 registry import cycle). Requires corpus handles so the four classes stay 

434 meaningfully distinct. 

435 """ 

436 meta = example_fixture_parameters(name, handles=handles) 

437 cap_max = meta["cap_max_rows"] 

438 assert isinstance(cap_max, int) 

439 out: dict[str, dict[str, object]] = {} 

440 for kind in ("positive", "empty", "tombstone", "cap"): 

441 params = meta[kind] 

442 assert isinstance(params, tuple) 

443 entry: dict[str, object] = {"parameters": params} 

444 if kind == "cap": 

445 entry["max_rows"] = cap_max 

446 out[kind] = entry 

447 return out
