13SURVIVOR_DEMAND_DDL = r""" 

14CREATE OR REPLACE VIEW v_graph_survivor (entity_id, survivor) AS 

15SELECT source.entity_id, terminal.cur 

16FROM entities AS source 

17CROSS JOIN LATERAL ( 

18 WITH RECURSIVE chain(cur, merged_into) AS ( 

19 SELECT source.entity_id, source.merged_into 

20 UNION 

21 SELECT next.entity_id, next.merged_into 

22 FROM chain 

23 JOIN entities AS next ON next.entity_id = chain.merged_into 

24 ) 

25 SELECT cur FROM chain WHERE merged_into IS NULL 

26) AS terminal; 

27"""

29ENTITIES_BOUNDED_PROVENANCE_DDL = r""" 

30CREATE OR REPLACE VIEW memory_v1.entities_current ( 

31 deployment_id, 

32 entity_id, 

33 entity_type, -- Vacated after D96: this compatibility output position always returns NULL because identity is the entity_id and no entity class is stored; use observation or profile fact text for kind-like retrieval. 

34 canonical_name, 

35 normalized_name, 

36 type_confidence, -- Vacated after D96: this compatibility output position always returns NULL because the removed entity-class vote has no confidence value; identity decisions are recorded by resolution tier and confidence instead. 

37 profile_summary, 

38 live_mention_count, 

39 live_document_count, 

40 graph_degree, 

41 created_at, 

42 updated_at 

43) AS 

44SELECT 

45 e.deployment_id, 

46 e.entity_id, 

47 NULL::text, 

48 e.canonical_name, 

49 e.normalized_name, 

50 NULL::real, 

51 e.profile_summary, 

52 live.mention_count, 

53 live.document_count, 

54 e.graph_degree::bigint, 

55 e.created_at, 

56 e.updated_at 

57FROM entities AS e 

58LEFT JOIN LATERAL ( 

59 SELECT coalesce(sum(edm.mention_count), 0)::bigint AS mention_count, 

60 count(*)::bigint AS document_count 

61 FROM memory_v1.entity_document_mentions AS edm 

62 WHERE edm.deployment_id = e.deployment_id 

63 AND edm.entity_id = e.entity_id 

64) AS live ON true 

65WHERE e.status = 'active' 

66 AND e.merged_into IS NULL 

67 AND EXISTS ( 

68 WITH RECURSIVE members(entity_id) AS ( 

69 SELECT e.entity_id 

70 UNION 

71 SELECT redirect.entity_id 

72 FROM entities AS redirect 

73 JOIN members ON redirect.merged_into = members.entity_id 

74 WHERE redirect.deployment_id = e.deployment_id 

75 ) 

76 SELECT 1 

77 FROM members 

78 WHERE EXISTS ( 

79 SELECT 1 FROM documents AS document 

80 WHERE document.deployment_id = e.deployment_id 

81 AND document.document_entity_id = members.entity_id 

82 AND document.deleted_at IS NULL 

83 ) OR EXISTS ( 

84 SELECT 1 

85 FROM LATERAL ( 

86 SELECT decision.deployment_id, decision.mention_id, decision.decision_id 

87 FROM resolution_decisions AS decision 

88 WHERE decision.deployment_id = e.deployment_id 

89 AND decision.entity_id = members.entity_id 

90 AND decision.superseded_by IS NULL 

91 OFFSET 0 

92 ) AS resolved 

93 JOIN mentions AS mention 

94 ON mention.deployment_id = resolved.deployment_id 

95 AND mention.mention_id = resolved.mention_id 

96 JOIN chunks AS chunk 

97 ON chunk.deployment_id = mention.deployment_id 

98 AND chunk.chunk_id = mention.chunk_id 

99 AND chunk.doc_id = mention.doc_id 

100 JOIN memory_v1.document_versions_visible AS version 

101 ON version.deployment_id = chunk.deployment_id 

102 AND version.version_id = chunk.version_id 

103 AND version.doc_id = chunk.doc_id 

104 JOIN document_representations AS representation 

105 ON representation.deployment_id = chunk.deployment_id 

106 AND representation.version_id = chunk.version_id 

107 AND representation.representation_id = chunk.representation_id 

108 AND representation.status = 'ready' 

109 WHERE resolved.decision_id = ( 

110 SELECT latest.decision_id 

111 FROM resolution_decisions AS latest 

112 WHERE latest.deployment_id = resolved.deployment_id 

113 AND latest.mention_id = resolved.mention_id 

114 AND latest.superseded_by IS NULL 

115 ORDER BY latest.decided_at DESC, latest.decision_id DESC 

116 LIMIT 1 

117 ) 

118 ) 

119 ); 

120 

121COMMENT ON VIEW memory_v1.entities_current IS 

122 'One row per externally visible survivor entity, keyed by (deployment_id, entity_id). Membership requires SURVIVING PROVENANCE, which is an explicit association to at least one live document lineage: a mention of this survivor in any non-tombstoned version of a live lineage, or a live document-entity bridge. An entity whose every source has been forgotten is therefore absent rather than orphaned, and merged entities are absent because a merge redirects to a survivor instead of rewriting history. MEMBERSHIP AND THE COUNTS ANSWER DIFFERENT QUESTIONS, and the difference is deliberate: the two counts are exact over CURRENT content only — they equal this entity''s rows in mentions_live and entity_document_mentions — so an entity whose only mention sits in a superseded version of a live lineage is published here with both counts at zero and has no row in entity_document_mentions at all. A zero count is not an absence of provenance. graph_degree is a deprecated compatibility scalar fixed at zero after D98 because current adjacency is computed from live PostgreSQL relations; profile_summary is orientation text, never evidence; and the clocks are registry maintenance instants that carry no world-validity meaning. After D96, entity_type and type_confidence are vacated (always NULL); identity is the entity_id.'; 

123 

124"""

126FACTS_DEMAND_COUNTS_DDL = r""" 

127CREATE OR REPLACE VIEW memory_v1.facts_visible_history ( 

128 deployment_id, -- The deployment that owns the fact. 

129 fact_kind, -- Which fact layer this row belongs to, either relation or observation. 

130 fact_id, -- Stable identity of the adjudicated fact. 

131 subject_entity_id, -- Survivor identity of the subject entity, with merge redirects resolved. 

132 predicate, -- The governed predicate of a relation, null for an observation. 

133 object_entity_id, -- Survivor identity of the object entity of a relation, null for an observation. 

134 statement, -- The canonical statement of an observation, null for a relation. 

135 fact_label, -- Human-readable sentence for the fact, null when no label has been generated. 

136 valid_from, -- Canonical inclusive world-time start; NULL means unknown. 

137 valid_until, -- Canonical exclusive world-time end; NULL means unknown unless valid_precision is open. 

138 ingested_at, -- Raw transaction-time start: when the system first believed the fact. 

139 invalidated_at, -- Raw transaction-time end: when the system learned the fact was superseded, null while it is still believed. 

140 contradiction_group, -- Shared identifier of an unadjudicated contradiction, null when the fact is in no contradiction group. 

141 confidence, -- Aggregate confidence over the fact's evidence, null when never scored. 

142 evidence_count_current, -- LIVE count of distinct current-testimony lineages supporting the fact, read now and never a historical reconstruction. 

143 contradict_count_current, -- LIVE count of distinct current-testimony lineages contradicting the fact, read now and never a historical reconstruction. 

144 support_state_current, -- LIVE support state, exactly current or withdrawn, derived now from the open review queue and never a stored column. 

145 valid_precision -- Chosen world-date precision; unknown and partial boundaries are distinct from explicitly open. 

146) AS 

147SELECT 

148 fact.deployment_id, 

149 fact.fact_kind, 

150 fact.fact_id, 

151 fact.subject_entity_id, 

152 fact.predicate, 

153 fact.object_entity_id, 

154 fact.statement, 

155 fact.fact_label, 

156 fact.valid_from, 

157 fact.valid_until, 

158 fact.ingested_at, 

159 fact.invalidated_at, 

160 fact.contradiction_group, 

161 fact.confidence, 

162 counts.supports, 

163 counts.contradicts, 

164 CASE WHEN EXISTS ( 

165 SELECT 1 

166 FROM review_queue AS queue 

167 WHERE queue.deployment_id = fact.deployment_id 

168 AND queue.item_kind = 'support_withdrawn' 

169 AND queue.status IN ('pending', 'deferred') 

170 AND queue.candidate ->> 'fact_kind' = fact.fact_kind 

171 AND queue.candidate ->> 'fact_id' = fact.fact_id::text 

172 ) THEN 'withdrawn' ELSE 'current' END, 

173 fact.valid_precision 

174FROM ( 

175 SELECT base.*, coalesce(relation.valid_precision, observation.valid_precision)::text AS valid_precision 

176 FROM v_memory_fact_visible AS base 

177 LEFT JOIN relations AS relation ON base.fact_kind='relation' 

178 AND relation.deployment_id=base.deployment_id AND relation.relation_id=base.fact_id 

179 LEFT JOIN observations AS observation ON base.fact_kind='observation' 

180 AND observation.deployment_id=base.deployment_id AND observation.observation_id=base.fact_id 

181) AS fact 

182LEFT JOIN LATERAL ( 

183 SELECT 

184 count(*) FILTER (WHERE lineage.stance = 'supports')::bigint AS supports, 

185 count(*) FILTER (WHERE lineage.stance = 'contradicts')::bigint AS contradicts 

186 FROM v_memory_evidence_lineage_live AS lineage 

187 WHERE lineage.deployment_id = fact.deployment_id 

188 AND lineage.fact_kind = fact.fact_kind 

189 AND lineage.fact_id = fact.fact_id 

190) AS counts ON true; 

191COMMENT ON VIEW memory_v1.facts_visible_history IS 'Historically visible adjudicated facts with one canonical world window and separate system timestamps. Membership requires surviving provenance. Unknown dates and partial windows are possible temporal matches, not proof of being current; open precision explicitly means ongoing. Completed windows remain believed history while invalidated_at is NULL. Evidence counts and support state are current testimony, not reconstructed historical counts.'; 

192"""
