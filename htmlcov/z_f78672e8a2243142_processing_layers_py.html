1"""D143 processing layers: configuration, boundaries, gaps and lineage order. 

2 

3A configured deployment (one with a ``deployment_processing_layers`` row) 

4decides at two **boundaries** whether a version enters a layer: 

5 

6- the claims boundary, when ``embed_chunk`` completes; 

7- the facts boundary, when the version's claim occurrences are complete (the 

8 extract barrier, or the empty path of a version without chunks). 

9 

10Every decision runs inside the completing transaction under the shared 

11**layer fence** and the D135 **lineage lock**, records a **gap** for the 

12version, and then runs one procedure — :func:`advance_on` — that enters the 

13lineage's next version once every earlier version reached the layer's reuse 

14**frontier**. The same procedure runs when a version reaches a frontier, when a 

15version is deleted and when ``setup`` seeds a layer that was switched on, so 

16the boundary and the backfill can never disagree. 

17 

18An unconfigured deployment never reaches this module's decision code: each 

19caller checks :func:`layer_configuration_on` first and keeps today's path. 

20"""

275def advance_on( 

276 *, connection: Connection, deployment_id: UUID, doc_id: UUID, layer: ProcessingLayer 

277) -> list[EnqueueOutcome]: 

278 """Enter the lineage's next version into ``layer`` when its order allows (§5.2). 

279 

280 The caller holds the layer fence and the lineage lock. Steps: 

281 

282 1. a disabled layer enters nothing (facts still opens the claims tails it 

283 now allows, whether or not claims are still enabled); 

284 2. a living lineage reopens its current version's discharged gap and 

285 discharges every other unentered gap; 

286 3. the target is the current version (living) or the oldest unentered gap 

287 (snapshot); 

288 4.–6. it is entered only when its sources layer is complete, every earlier 

289 version reached the frontier (or was discharged), and — for facts — its 

290 claim occurrences are complete. 

291 """ 

292 configuration = layer_configuration_on( 

293 connection=connection, deployment_id=deployment_id 

294 ) 

295 lineage = _read_lineage( 

296 connection=connection, deployment_id=deployment_id, doc_id=doc_id 

297 ) 

298 if lineage.living and _reopen_current_gaps( 

299 connection=connection, lineage=lineage, layer=layer 

300 ): 

301 lineage = _read_lineage( 

302 connection=connection, deployment_id=deployment_id, doc_id=doc_id 

303 ) 

304 if not configuration.enabled(layer=layer): 

305 if layer is ProcessingLayer.FACTS: 

306 # §4.4: claims work that already entered drains through its tail 

307 # even when claims were switched off meanwhile. 

308 return _open_owed_claims_tails( 

309 connection=connection, 

310 lineage=lineage, 

311 facts_enabled=False, 

312 admitted=None, 

313 ) 

314 return [] 

315 if lineage.living: 

316 _discharge_superseded_gaps(connection=connection, lineage=lineage, layer=layer) 

317 lineage = _read_lineage( 

318 connection=connection, deployment_id=deployment_id, doc_id=doc_id 

319 ) 

320 target = lineage.target(layer=layer) 

321 admitted = ( 

322 target 

323 if target is not None and lineage.admits(target=target, layer=layer) 

324 else None 

325 ) 

326 outcomes: list[EnqueueOutcome] = [] 

327 if layer is ProcessingLayer.FACTS: 

328 outcomes.extend( 

329 _open_owed_claims_tails( 

330 connection=connection, 

331 lineage=lineage, 

332 facts_enabled=True, 

333 admitted=None if admitted is None else admitted.version_id, 

334 ) 

335 ) 

336 if admitted is None: 

337 return outcomes 

338 connection.execute( 

339 _STAMP_ENTERED, 

340 { 

341 "deployment_id": deployment_id, 

342 "version_id": admitted.version_id, 

343 "layer": layer, 

344 }, 

345 ) 

346 # A connector cycle that finalized while this version waited outside the 

347 # layer must evaluate its closure again once the resumed work is done. 

348 connection.execute(_REOPEN_FINALIZED_CYCLE, {"version_id": admitted.version_id}) 

349 lane = admitted.gaps[layer].lane 

350 if layer is ProcessingLayer.CLAIMS: 

351 outcomes.extend( 

352 _enter_claims( 

353 connection=connection, 

354 deployment_id=deployment_id, 

355 version_id=admitted.version_id, 

356 lane=lane, 

357 ) 

358 ) 

359 else: 

360 outcomes.extend( 

361 _enter_facts( 

362 connection=connection, 

363 deployment_id=deployment_id, 

364 version_id=admitted.version_id, 

365 lane=lane, 

366 ) 

367 ) 

368 return outcomes

459 """Advance every lineage with an open unentered gap of an enabled layer. 

460 

461 Runs one bounded transaction per lineage and layer, in batches, and is

463 lineage advances ran.

542def admit_layers_on( 

543 *, 

544 connection: Connection, 

545 deployment_id: UUID, 

546 channels: frozenset[ProcessingLayer], 

547 requested: frozenset[ProcessingLayer] | None, 

548 mandatory: frozenset[ProcessingLayer] = frozenset(), 

549) -> LayerAdmission: 

550 """Apply the D143 §7.2 rules to one read. 

551 

552 A requested or mandatory layer the deployment does not serve is 

553 ``LayerUnavailableError`` (409); a mandatory layer the request excluded, or 

554 a selection leaving none of the read's channels, is ``LayerSelectionError`` 

555 (422). A deployment without a configuration row serves everything, so an 

556 unconfigured read with no selection consults every channel unchanged. 

557 """ 

558 states = layer_states_on(connection=connection, deployment_id=deployment_id) 

559 

560 def served(layer: ProcessingLayer) -> bool: 

561 if layer is ProcessingLayer.CLAIMS: 

562 return states.claims != "disabled" 

563 if layer is ProcessingLayer.FACTS: 

564 return states.facts != "disabled" 

565 return True 

566 

567 def backfilling(layer: ProcessingLayer) -> bool: 

568 if layer is ProcessingLayer.CLAIMS: 

569 return states.claims == "backfilling" 

570 if layer is ProcessingLayer.FACTS: 

571 return states.facts == "backfilling" 

572 return False 

573 

574 for layer in sorted(requested or (), key=lambda item: item.value): 

575 if not served(layer): 

576 raise LayerUnavailableError( 

577 layer=layer, 

578 message=f"this deployment does not process the {layer.value} layer", 

579 ) 

580 for layer in sorted(mandatory, key=lambda item: item.value): 

581 if not served(layer): 

582 raise LayerUnavailableError( 

583 layer=layer, 

584 message=( 

585 f"this request needs the {layer.value} layer, which this" 

586 " deployment does not process" 

587 ), 

588 ) 

589 if requested is not None and layer not in requested: 

590 raise LayerSelectionError( 

591 layer=layer, 

592 message=( 

593 f"this request needs the {layer.value} layer, which the" 

594 " layers selection excludes" 

595 ), 

596 ) 

597 selected = channels if requested is None else channels & requested 

598 consulted = frozenset(layer for layer in selected if served(layer)) 

599 if not consulted: 

600 if requested is not None and not selected: 

601 raise LayerSelectionError( 

602 layer=None, 

603 message="the layers selection excludes every layer this read uses", 

604 ) 

605 layer = sorted(channels, key=lambda item: item.value)[0] 

606 raise LayerUnavailableError( 

607 layer=layer, 

608 message=f"this deployment does not process the {layer.value} layer", 

609 ) 

610 return LayerAdmission( 

611 consulted=consulted, 

612 unconsulted=channels - consulted, 

613 backfilling=frozenset(layer for layer in consulted if backfilling(layer)), 

614 )
