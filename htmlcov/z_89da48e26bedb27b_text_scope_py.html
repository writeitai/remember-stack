1"""D140 time scope of a text read, compiled to SQL predicates (§3.2, §8.1). 

2 

3Every read that returns document text, text-derived claims or facts under a 

4``time`` argument (``current``, ``at``, ``overlap``, ``history``) selects the 

5document versions in force for that scope. The rule lives once in SQL as 

6``memory_v1.versions_in_scope``; ranked statements cannot call a set-returning 

7function per candidate, so this module renders the same rule as a 

8primary-key probe on the current-belief projection ``document_version_scope`` 

9that sits *inside* the ranked statement, before its ``ORDER BY … LIMIT``: 

10 

11- a version is **selected** when its projection row is ``selectable`` (ready, 

12 with a ready current reading) and its in-force multirange overlaps the 

13 scope's window. A lineage without declared periods has its served version 

14 in the projection with an unbounded range, so the one probe covers every 

15 lineage and an undeclared corpus reads exactly as before; 

16- a **claim** of a periodised lineage is selected through any occurrence in 

17 a selected version; a claim of an undeclared lineage keeps today's 

18 current-testimony rule; 

19- a **fact** passes the evidence gate when a supporting claim occurs in a 

20 selected version or in any live version of an undeclared lineage. A fact 

21 with no live supporting occurrence at all rests on no document text, so 

22 the gate does not apply to it. 

23 

24The window is the inclusive instant range the mode stands for: ``current`` 

25is the evaluation instant, ``at`` one instant, ``overlap`` its inclusive 

26bounds and ``history`` everything up to the evaluation instant. Every 

27statement that uses these fragments binds ``:deployment_id``. 

28"""

256def fact_in_scope(*, fact_kind: str, fact_id: str, pinned_belief: bool = False) -> str: 

257 """One predicate: the §8.1 evidence gate for one fact of a fixed kind. 

258 

259 ``document_version_scope`` holds exactly the non-deleted versions of live 

260 lineages, so joining it is the "live occurrence" test. An occurrence is in 

261 scope when it lies in a version the scope selects, or in any live version 

262 of a lineage without declared periods (whose evidence D140 does not 

263 time-restrict), and in that version's current reading. 

264 

265 A fact with supporting evidence needs an in-scope *supporting* occurrence; 

266 support left only in deleted versions or replaced readings does not 

267 count. A fact with no supporting evidence at all — a D54 zero-support, 

268 contradiction-only fact, which D54 flags rather than hides — needs an 

269 in-scope occurrence of any stance, so such facts read exactly as before in 

270 a corpus without declared periods and are still time-restricted in one 

271 with them. For current belief the selection is the projection probe; for 

272 a past belief instant the fact's bounded evidence lineages are evaluated 

273 from the ledgers through ``memory_v1.fact_in_scope_support``. 

274 """ 

275 if fact_kind not in _EVIDENCE: 

276 raise ValueError(f"unknown fact kind {fact_kind!r}") 

277 alias = f"gate_{fact_kind}" 

278 table, column = _EVIDENCE[fact_kind]

280 def in_scope_occurrence(*, stance: str | None, suffix: str) -> str: 

281 scoped = f"{alias}{suffix}" 

282 occurrences = _support_occurrences( 

283 fact_kind=fact_kind, fact_id=fact_id, alias=scoped, stance=stance 

284 ) 

285 return ( 

286 f"EXISTS (SELECT 1 {occurrences}" 

287 f" AND (NOT {scoped}_scope.periodised" 

288 f" OR ({scoped}_scope.selectable" 

289 f" AND {scoped}_scope.in_force && {WINDOW_SQL})))" 

290 ) 

291 

292 if not pinned_belief: 

293 unsupported = ( 

294 f"NOT EXISTS (SELECT 1 FROM {table} {alias}_any" 

295 f" WHERE {alias}_any.deployment_id = :deployment_id" 

296 f" AND {alias}_any.{column} = {fact_id}" 

297 f" AND {alias}_any.stance = 'supports')" 

298 ) 

299 in_scope = ( 

300 f"{in_scope_occurrence(stance='supports', suffix='')}" 

301 f" OR ({unsupported}" 

302 f" AND {in_scope_occurrence(stance=None, suffix='_all')})" 

303 ) 

304 else: 

305 in_scope = ( 

306 "memory_v1.fact_in_scope_support(" 

307 f":deployment_id, '{fact_kind}', {fact_id}," 

308 " CAST(:scope_mode AS text), CAST(:scope_at AS timestamptz)," 

309 " CAST(:scope_range_start AS timestamptz)," 

310 " CAST(:scope_range_end AS timestamptz)," 

311 " CAST(:scope_evaluated_at AS timestamptz)," 

312 " CAST(:scope_believed_at AS timestamptz))" 

313 ) 

314 return f"({in_scope})"

317def fact_in_scope_by_kind(*, kind: str, fact_id: str) -> str:

329def claim_in_scope(*, claim: str, alias: str = "evidence_scope") -> str: 

330 """One predicate: claim ``claim`` may be shown as fact evidence (§8.1). 

331 

332 The claim occurs, in a version's current reading, in a selected version or 

333 in a live version of an undeclared lineage. The same rule as the fact 

334 gate, at claim grain (current belief): an occurrence left only in a 

335 replaced reading or a deleted version is not evidence. 

336 """ 

337 occurrences = ( 

338 f"FROM public.chunk_claims {alias}" 

339 f" JOIN public.chunks {alias}_chunk" 

340 f" ON {alias}_chunk.deployment_id = {alias}.deployment_id" 

341 f" AND {alias}_chunk.chunk_id = {alias}.chunk_id" 

342 f"{_current_reading(chunk=f'{alias}_chunk', alias=f'{alias}_version')}" 

343 f" JOIN public.document_version_scope {alias}_scope" 

344 f" ON {alias}_scope.deployment_id = {alias}_chunk.deployment_id" 

345 f" AND {alias}_scope.version_id = {alias}_chunk.version_id" 

346 f" WHERE {alias}.deployment_id = :deployment_id" 

347 f" AND {alias}.claim_id = {claim}" 

348 ) 

349 return ( 

350 f"EXISTS (SELECT 1 {occurrences}" 

351 f" AND (NOT {alias}_scope.periodised" 

352 f" OR ({alias}_scope.selectable AND {alias}_scope.in_force && {WINDOW_SQL})))" 

353 )
