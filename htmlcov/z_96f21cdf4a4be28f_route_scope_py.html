1"""Which perimeter scope each route requires. 

2 

3## Why this is a table and not a rule 

4 

5The obvious rule — "GET reads, POST writes" — is wrong about this API, and 

6believing it would hand a read-only credential the ability to ingest. 

7``POST /graph/neighborhood``, ``POST /graph/path``, ``POST /query/sql`` and 

8``POST /readiness`` are reads that use POST because their arguments do not fit 

9in a query string. ``POST /ingest`` and ``POST /connectors`` change the 

10memory. The method tells you nothing about which is which. 

11 

12So the mapping is enumerated by hand, and the default for anything not 

13enumerated is ``WRITE``. 

14 

15## The direction failure runs 

16 

17An unmapped route requires the *stronger* scope, so adding a route without 

18classifying it denies read-only callers a feature. The opposite default would 

19grant them a write. One of those is a bug report; the other is an incident. 

20 

21``POST /operations/{name}`` cannot be classified from the path — operations are 

22registry data, and which of them mutate is a property of the operation. 

23:func:`operation_scope` therefore asks the descriptor, and an operation that has 

24not declared itself non-mutating requires ``WRITE``. Absence is refusal, never 

25permission. 

26""" 

27

34#: Routes a read-only credential may reach, as (method, path regex). 

35#: 

36#: Anchored patterns, because a prefix match would let ``/search/claims/../..`` 

37#: style path games widen the set. The path is matched after normalisation by 

38#: the router, so these mirror the declared routes exactly. 

39_READ_ROUTES: tuple[tuple[str, re.Pattern[str]], ...] = tuple( 

40 (method, re.compile(pattern)) 

41 for method, pattern in ( 

42 ("GET", r"^/healthz$"), 

43 ("GET", r"^/resolve$"), 

44 ("GET", r"^/lookup/relations$"), 

45 ("GET", r"^/lookup/observations$"), 

46 ("GET", r"^/transcript/relation/[^/]+$"), 

47 ("GET", r"^/hydrate/relation/[^/]+$"), 

48 ("GET", r"^/search/claims$"), 

49 ("GET", r"^/search/chunks$"), 

50 # POST, and still a read: the argument shape does not fit a query 

51 # string. This is exactly the case the method-based rule gets wrong. 

52 ("POST", r"^/graph/neighborhood$"), 

53 ("POST", r"^/graph/path$"), 

54 ("POST", r"^/graph/citation-path$"), 

55 ("POST", r"^/query/sql$"), 

56 ("POST", r"^/query/sql/explain$"), 

57 ("GET", r"^/query/space$"), 

58 ("GET", r"^/query/space/search$"), 

59 ("GET", r"^/query/saved$"), 

60 ("GET", r"^/query/saved/[^/]+/[^/]+$"), 

61 ("POST", r"^/query/saved/[^/]+/[^/]+/run$"), 

62 ("POST", r"^/readiness$"), 

63 ("GET", r"^/operations$"), 

64 ("GET", r"^/connectors$"), 

65 ("GET", r"^/connectors/[^/]+$"), 

66 ) 

67) 

68

70#: Routes whose scope cannot be decided from the path, and which enforce it 

71#: themselves. Exactly one today: an assured operation's authority is a 

72#: property of the operation, and operations are registry data. 

73#: 

74#: A route listed here **must** perform its own check — the perimeter has 

75#: deliberately stood aside. That is a real hazard, so the set is closed, tiny, 

76#: and asserted on in tests rather than left as a convention. 

77_ROUTE_DECIDES: tuple[tuple[str, re.Pattern[str]], ...] = ( 

78 ("POST", re.compile(r"^/operations/[^/]+$")), 

79) 

80 

81 

82def required_scope(*, method: str, path: str) -> PerimeterScope | None: 

83 """The scope a caller needs for this route. 

84 

85 ``None`` means the route decides for itself (see :data:`_ROUTE_DECIDES`); 

86 the perimeter enforces nothing and the handler must. 

87 

88 Unenumerated routes require :attr:`PerimeterScope.WRITE`, so a route added 

89 without a decision here is closed to read-only callers rather than open. 

90 """ 

91 normalised = path.rstrip("/") or "/" 

92 upper = method.upper() 

93 for route_method, pattern in _ROUTE_DECIDES: 

94 if route_method == upper and pattern.match(normalised): 

95 return None 

96 for route_method, pattern in _READ_ROUTES: 

97 if route_method == upper and pattern.match(normalised): 

98 return PerimeterScope.READ 

99 return PerimeterScope.WRITE 

100 

101 

102def routes_that_decide_for_themselves() -> tuple[tuple[str, str], ...]: 

103 """The deferring routes, for tests that pin the set to exactly what is intended.""" 

104 return tuple((method, pattern.pattern) for method, pattern in _ROUTE_DECIDES) 

105

107def operation_scope(*, mutates: bool | None) -> PerimeterScope: 

108 """The scope an assured operation requires. 

109 

110 ``None`` means the descriptor has not said, which is refused for read-only 

111 callers rather than assumed harmless: an operation nobody classified is an 

112 operation nobody has checked. 

113 """ 

114 if mutates is False: 

115 return PerimeterScope.READ 

116 return PerimeterScope.WRITE
