1"""Which perimeter scope each route requires. 

2 

3## Why this is a table and not a rule 

4 

5The obvious rule — "GET reads, POST writes" — is wrong about this API, and 

6believing it would hand a read-only credential the ability to ingest. 

7``POST /graph/neighborhood``, ``POST /graph/path``, ``POST /query/sql`` and 

8``POST /readiness`` are reads that use POST because their arguments do not fit 

9in a query string. ``POST /ingest`` and ``POST /connectors`` change the memory, 

10but D62 permits a browser to do only the first. The method tells you nothing 

11about which is which. 

12 

13So the mapping is enumerated by hand, and the default for anything not 

14enumerated is ``WRITE``. 

15 

16## The direction failure runs 

17 

18An unmapped route requires the *stronger* scope, so adding a route without 

19classifying it denies read-only callers a feature. The opposite default would 

20grant them a write. One of those is a bug report; the other is an incident. 

21 

22``POST /operations/{name}`` cannot be classified from the path — operations are 

23registry data, and which of them mutate is a property of the operation. 

24:func:`operation_scope` therefore asks the descriptor, and an operation that has 

25not declared itself non-mutating requires ``WRITE``. Absence is refusal, never 

26permission. 

27""" 

28

35#: Routes a read-only credential may reach, as (method, path regex). 

36#: 

37#: Anchored patterns, because a prefix match would let ``/search/claims/../..`` 

38#: style path games widen the set. The path is matched after normalisation by 

39#: the router, so these mirror the declared routes exactly. 

40_READ_ROUTES: tuple[tuple[str, re.Pattern[str]], ...] = tuple( 

41 (method, re.compile(pattern)) 

42 for method, pattern in ( 

43 ("GET", r"^/healthz$"), 

44 ("GET", r"^/resolve$"), 

45 ("GET", r"^/lookup/relations$"), 

46 ("GET", r"^/lookup/observations$"), 

47 ("GET", r"^/transcript/relation/[^/]+$"), 

48 ("GET", r"^/hydrate/relation/[^/]+$"), 

49 ("GET", r"^/search/claims$"), 

50 ("GET", r"^/search/chunks$"), 

51 # The body-carrying forms of the same two reads. A search does not 

52 # become a write by moving its terms out of the request line (D59). 

53 ("POST", r"^/search/claims$"), 

54 ("POST", r"^/search/chunks$"), 

55 # POST, and still a read: the argument shape does not fit a query 

56 # string. This is exactly the case the method-based rule gets wrong. 

57 ("POST", r"^/graph/neighborhood$"), 

58 ("POST", r"^/graph/path$"), 

59 ("POST", r"^/graph/citation-path$"), 

60 ("POST", r"^/query/sql$"), 

61 ("POST", r"^/query/sql/explain$"), 

62 ("GET", r"^/query/space$"), 

63 ("GET", r"^/query/space/search$"), 

64 ("GET", r"^/query/saved$"), 

65 ("GET", r"^/query/saved/[^/]+/[^/]+$"), 

66 ("POST", r"^/query/saved/[^/]+/[^/]+/run$"), 

67 ("POST", r"^/readiness$"), 

68 ("GET", r"^/operations$"), 

69 ("GET", r"^/connectors$"), 

70 ("GET", r"^/connectors/[^/]+$"), 

71 # The inventory of what the deployment holds. A read, and the one a 

72 # browser credential needs most: without it the app can show counts 

73 # but never which document they refer to. 

74 ("GET", r"^/documents$"), 

75 ) 

76)

79#: The single route a narrow D62 browser ingest credential may reach. 

80#: 

81#: Kept separate from reads because an ingest credential cannot read memory, 

82#: and separate from the WRITE fallback because it must not create standing 

83#: connectors or perform any other mutation. 

84_INGEST_ROUTES: tuple[tuple[str, re.Pattern[str]], ...] = ( 

85 ("POST", re.compile(r"^/ingest$")), 

86) 

87 

88 

89#: Routes whose scope cannot be decided from the path, and which enforce it 

90#: themselves. Exactly one today: an assured operation's authority is a 

91#: property of the operation, and operations are registry data. 

92#: 

93#: A route listed here **must** perform its own check — the perimeter has 

94#: deliberately stood aside. That is a real hazard, so the set is closed, tiny, 

95#: and asserted on in tests rather than left as a convention. 

96_ROUTE_DECIDES: tuple[tuple[str, re.Pattern[str]], ...] = ( 

97 ("POST", re.compile(r"^/operations/[^/]+$")), 

98)

101def required_scope(*, method: str, path: str) -> PerimeterScope | None: 

102 """The scope a caller needs for this route. 

103 

104 ``None`` means the route decides for itself (see :data:`_ROUTE_DECIDES`); 

105 the perimeter enforces nothing and the handler must. 

106 

107 Unenumerated routes require :attr:`PerimeterScope.WRITE`, so a route added 

108 without a decision here is closed to narrow credentials rather than open. 

109 """ 

110 normalised = path.rstrip("/") or "/" 

111 upper = method.upper() 

112 for route_method, pattern in _ROUTE_DECIDES: 

113 if route_method == upper and pattern.match(normalised): 

114 return None 

115 for route_method, pattern in _INGEST_ROUTES: 

116 if route_method == upper and pattern.match(normalised): 

117 return PerimeterScope.INGEST 

118 for route_method, pattern in _READ_ROUTES: 

119 if route_method == upper and pattern.match(normalised): 

120 return PerimeterScope.READ 

121 return PerimeterScope.WRITE 

122 

123 

124def routes_that_decide_for_themselves() -> tuple[tuple[str, str], ...]: 

125 """The deferring routes, for tests that pin the set to exactly what is intended.""" 

126 return tuple((method, pattern.pattern) for method, pattern in _ROUTE_DECIDES)

129def operation_scope(*, mutates: bool | None) -> PerimeterScope: 

130 """The scope an assured operation requires. 

131 

132 ``None`` means the descriptor has not said, which is refused for read-only 

133 callers rather than assumed harmless: an operation nobody classified is an 

134 operation nobody has checked. 

135 """ 

136 if mutates is False: 

137 return PerimeterScope.READ 

138 return PerimeterScope.WRITE
