1"""Opt-in full-payload generation recorder for benchmark diagnosis. 

2 

3R13 post-mortem: 29 dead letters were inexplicable because the rejected 

4response bytes were never stored — adapter error paths deliberately log 

5metadata only (lengths, digests). This module is the explicit, env-gated 

6exception: when ``REMEMBERSTACK_LANGFUSE_ENABLED=1`` with endpoint and 

7project keys set, the Vertex and OpenRouter adapters report every 

8generation — the full prompt, the full raw completion, usage, and outcome. 

9 

10Transport, proven 2026-09-16: plain OpenTelemetry spans over OTLP to the 

11obs ingest path, with ``gen_ai.prompt`` / ``gen_ai.completion`` carrying 

12the full payloads (the v4 Langfuse SDK's generations never surfaced on the 

13v3 self-hosted server, while an identical OTLP span round-tripped fine). 

14 

15Default off: :func:`build_generation_recorder` returns ``None`` unless the 

16switch is enabled, so deployments that never set these variables behave 

17exactly as before. Enabled without endpoint or keys fails loud at startup 

18instead of running blind. The recorder never breaks a generation: emit 

19failures are logged, never raised. 

20"""

148def build_generation_recorder( 

149 *, settings: LangfuseRecorderSettings 

150) -> GenerationRecorder | None: 

151 """Return a recorder when explicitly enabled, else ``None``. 

152 

153 Enabled without endpoint or project keys raises instead of running 

154 blind: a benchmark that asked for tracking must not silently produce 

155 none. 

156 """ 

157 if not settings.enabled: 

158 return None 

159 host = settings.host.strip() 

160 if not host: 

161 raise ValueError("REMEMBERSTACK_LANGFUSE_HOST is required when enabled") 

162 public_key = ( 

163 "" 

164 if settings.public_key is None 

165 else settings.public_key.get_secret_value().strip() 

166 ) 

167 secret_key = ( 

168 "" 

169 if settings.secret_key is None 

170 else settings.secret_key.get_secret_value().strip() 

171 ) 

172 if not public_key or not secret_key: 

173 raise ValueError( 

174 "REMEMBERSTACK_LANGFUSE_PUBLIC_KEY and" 

175 " REMEMBERSTACK_LANGFUSE_SECRET_KEY are required when enabled" 

176 ) 

177 tracer, flusher, trace_api = _build_otel_tracer( 

178 endpoint=host.rstrip("/") + _OTLP_TRACES_PATH, 

179 public_key=public_key, 

180 secret_key=secret_key, 

181 ca_file=settings.ca_file.strip() or None, 

182 ) 

183 return OtelSpanRecorder( 

184 tracer=tracer, 

185 flusher=flusher, 

186 trace_api=trace_api, 

187 run_tag=settings.run_tag.strip(), 

188 )

202def _build_otel_tracer( 

203 *, endpoint: str, public_key: str, secret_key: str, ca_file: str | None 

204) -> tuple[_Tracer, Callable[[], None], Any]: 

205 """Build one OTLP tracer plus its flush callable. 

206 

207 The SDK is resolved with ``import_module`` (like ``adapters/sentry.py``) 

208 so static analysis and runtimes without the optional ``observability`` 

209 extra never touch a missing import. 

210 """ 

211 sdk_trace = _load_module( 

212 name="opentelemetry.sdk.trace", package="opentelemetry-sdk" 

213 ) 

214 sdk_export = _load_module( 

215 name="opentelemetry.sdk.trace.export", package="opentelemetry-sdk" 

216 ) 

217 otlp_http = _load_module( 

218 name="opentelemetry.exporter.otlp.proto.http.trace_exporter", 

219 package="opentelemetry-exporter-otlp-proto-http", 

220 ) 

221 credentials = base64.b64encode(f"{public_key}:{secret_key}".encode()).decode() 

222 exporter = otlp_http.OTLPSpanExporter( 

223 endpoint=endpoint, 

224 headers={"Authorization": f"Basic {credentials}"}, 

225 certificate_file=ca_file, 

226 ) 

227 provider = sdk_trace.TracerProvider() 

228 provider.add_span_processor(sdk_export.BatchSpanProcessor(exporter)) 

229 tracer = provider.get_tracer("rememberstack.generation_recorder") 

230 trace_api = _load_module(name="opentelemetry.trace", package="opentelemetry-sdk") 

231 return tracer, provider.force_flush, trace_api
