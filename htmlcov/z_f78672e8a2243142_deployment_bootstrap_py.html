186class DeploymentBootstrapper: 

187 """Create or verify one D68 deployment and its exact core-v1 manifest."""

189 def __init__(self, *, engine: Engine) -> None: 

190 """Bind bootstrap to an explicitly composed SQLAlchemy engine.""" 

191 self._engine = engine 

192 

193 def bootstrap_deployment( 

194 self, *, deployment_input: DeploymentBootstrapInput 

195 ) -> DeploymentBootstrapResult: 

196 """Create or verify all bootstrap state in one owned transaction.""" 

197 with self._engine.begin() as connection: 

198 connection.execute(statement=text(_LOCK_DEPLOYMENT_BOOTSTRAP)) 

199 deployment_created = _create_or_compare_deployment( 

200 connection=connection, deployment_input=deployment_input 

201 ) 

202 if deployment_created: 

203 _insert_core_manifest( 

204 connection=connection, deployment_id=deployment_input.deployment_id 

205 ) 

206 else: 

207 _compare_core_manifest( 

208 connection=connection, deployment_id=deployment_input.deployment_id 

209 ) 

210 

211 return DeploymentBootstrapResult( 

212 deployment_id=deployment_input.deployment_id, 

213 deployment_created=deployment_created, 

214 entity_types_count=len(CORE_MANIFEST.entity_types), 

215 predicates_count=len(CORE_MANIFEST.predicates), 

216 predicate_signatures_count=len(CORE_MANIFEST.predicate_signatures), 

217 )

220def _create_or_compare_deployment( 

221 *, connection: Connection, deployment_input: DeploymentBootstrapInput 

222) -> bool: 

223 """Insert an absent deployment or compare every mapped input field exactly.""" 

224 expected = _deployment_values(deployment_input=deployment_input) 

225 rows = ( 

226 connection.execute( 

227 statement=text(_SELECT_DEPLOYMENT), 

228 parameters={ 

229 "deployment_id": deployment_input.deployment_id, 

230 "slug": deployment_input.slug, 

231 }, 

232 ) 

233 .mappings() 

234 .all() 

235 ) 

236 if not rows: 

237 connection.execute( 

238 statement=text(_INSERT_DEPLOYMENT), parameters=dict(expected) 

239 ) 

240 return True 

241 if len(rows) != 1 or dict(rows[0]) != expected: 

242 raise DeploymentConflictError( 

243 "deployment identity or mapped profile values conflict" 

244 ) 

245 return False

248def _insert_core_manifest(*, connection: Connection, deployment_id: UUID) -> None: 

249 """Insert the complete manifest in its binding foreign-key order.""" 

250 connection.execute( 

251 statement=text(_INSERT_ENTITY_TYPE), 

252 parameters=[ 

253 _entity_type_values(deployment_id=deployment_id, definition=definition) 

254 for definition in CORE_MANIFEST.entity_types 

255 ], 

256 ) 

257 connection.execute( 

258 statement=text(_INSERT_PREDICATE), 

259 parameters=[ 

260 _predicate_values(deployment_id=deployment_id, definition=definition) 

261 for definition in CORE_MANIFEST.predicates 

262 ], 

263 ) 

264 connection.execute( 

265 statement=text(_INSERT_PREDICATE_SIGNATURE), 

266 parameters=[ 

267 _predicate_signature_values( 

268 deployment_id=deployment_id, definition=definition 

269 ) 

270 for definition in CORE_MANIFEST.predicate_signatures 

271 ], 

272 )

275def _compare_core_manifest(*, connection: Connection, deployment_id: UUID) -> None: 

276 """Compare the complete stored core state without mutating a valid retry.""" 

277 parameters = {"deployment_id": deployment_id} 

278 entity_rows = ( 

279 connection.execute( 

280 statement=text(_SELECT_CORE_ENTITY_TYPES), parameters=parameters 

281 ) 

282 .mappings() 

283 .all() 

284 ) 

285 predicate_rows = ( 

286 connection.execute( 

287 statement=text(_SELECT_CORE_PREDICATES), parameters=parameters 

288 ) 

289 .mappings() 

290 .all() 

291 ) 

292 signature_rows = ( 

293 connection.execute( 

294 statement=text(_SELECT_CORE_PREDICATE_SIGNATURES), parameters=parameters 

295 ) 

296 .mappings() 

297 .all() 

298 ) 

299 

300 if _entity_state(rows=entity_rows) != _expected_entity_state(): 

301 raise CoreManifestConflictError("core entity-type state conflicts") 

302 if _predicate_state(rows=predicate_rows) != _expected_predicate_state(): 

303 raise CoreManifestConflictError("core predicate state conflicts") 

304 if _signature_state(rows=signature_rows) != _expected_signature_state(): 

305 raise CoreManifestConflictError("core predicate-signature state conflicts")

405def _predicate_state(*, rows: Sequence[RowMapping]) -> dict[str, dict[str, object]]: 

406 """Canonicalize stored predicates while validating mutable counters.""" 

407 state: dict[str, dict[str, object]] = {} 

408 for row in rows: 

409 usage_count = row["usage_count"] 

410 if not isinstance(usage_count, int) or usage_count < 0: 

411 raise CoreManifestConflictError("core predicate usage_count is invalid") 

412 state[str(row["predicate"])] = { 

413 key: value 

414 for key, value in row.items() 

415 if key not in {"predicate", "usage_count"} 

416 } 

417 return state
