1"""D134 general document metadata: per-version fields, people, and observed names. 

2 

3Existing versions are backfilled from what the spine already records: the 

4family from the stored MIME (the same mapping as 

5``rememberstack.core.document_metadata.family_for_mime``) and the version's 

6language, thread key and source dates. What title the caller declared was 

7never recorded, so ``document_metadata.title`` stays NULL (a later 

8conversion can fill it); the lineage title becomes the version's first 

9``document_names`` row, so the document stays findable by it. File names and 

10source paths were never recorded either, so they stay NULL. Lineages with a 

11hard-forget manifest are skipped: their source-bearing rows were scrubbed and 

12must not be re-derived. 

13 

14The backfill is one statement in the migration transaction and holds locks 

15on ``document_versions`` while it runs; a keyset backfill (as p9_16 does) is 

16the path if a deployment is too large for that. 

17 

18The downgrade refuses to drop populated tables: metadata observed at ingest 

19(file names, paths, observed names) cannot be re-derived from anything else. 

20"""

37_DDL = r""" 

38CREATE TABLE document_metadata ( 

39 deployment_id uuid NOT NULL, 

40 version_id uuid NOT NULL, 

41 doc_id uuid NOT NULL, -- denormalized lineage for filters 

42 family text NOT NULL, -- D133 format family 

43 file_name text, -- as observed for this version 

44 source_path text, -- the source location as observed for this version (lineage source_uri is mutable) 

45 title text, -- the title the document declares 

46 created_at timestamptz, -- source-declared created/sent 

47 modified_at timestamptz, -- source-declared last modified 

48 language text, -- detected primary language 

49 thread_ref text, -- opaque conversation/thread key 

50 provenance jsonb NOT NULL DEFAULT '{}', -- field → source | connector 

51 extra jsonb NOT NULL DEFAULT '{}', -- family-specific fields; returned, not a general filter 

52 metadata_mapping_version text NOT NULL, -- the family's metadata mapping (converter/connector), distinct from the E2 extractor version 

53 PRIMARY KEY (deployment_id, version_id), 

54 FOREIGN KEY (deployment_id, version_id) REFERENCES document_versions (deployment_id, version_id) ON DELETE CASCADE 

55); 

56COMMENT ON TABLE document_metadata IS 

57 'D134 general document metadata: one row per version, the same fields for every format family. Hard forget deletes the lineage''s rows.'; 

58CREATE INDEX ix_document_metadata_family ON document_metadata (deployment_id, family, created_at DESC); 

59CREATE INDEX ix_document_metadata_thread ON document_metadata (deployment_id, thread_ref) WHERE thread_ref IS NOT NULL; 

60 

61CREATE TABLE document_names ( 

62 deployment_id uuid NOT NULL, 

63 version_id uuid NOT NULL, 

64 observed_at timestamptz NOT NULL, -- when this name was observed for the version 

65 file_name text, -- the file name observed 

66 title text, -- the declared title observed 

67 source_path text, -- the source location observed 

68 name_text text NOT NULL, -- file_name + title + source_path, space-joined; the indexed search text 

69 origin text NOT NULL CHECK (origin IN ('ingest','observation','converter','backfill')), -- ingest | observation (same bytes, new name) | converter (declared by the file) | backfill (legacy lineage title) 

70 PRIMARY KEY (deployment_id, version_id, observed_at), 

71 FOREIGN KEY (deployment_id, version_id) REFERENCES document_metadata (deployment_id, version_id) ON DELETE CASCADE 

72); 

73COMMENT ON TABLE document_names IS 

74 'D134 every name a version was observed under: the name at ingest, then one row per metadata observation (identical bytes under a new file name, title or path). Hard forget deletes the lineage''s rows.'; 

75CREATE INDEX ix_document_names_trgm ON document_names USING gin (name_text gin_trgm_ops); 

76CREATE INDEX ix_document_names_bm25 ON document_names USING bm25 (name_text) WITH (text_config='simple'); 

77 

78CREATE TABLE document_people ( 

79 deployment_id uuid NOT NULL, 

80 version_id uuid NOT NULL, 

81 role text NOT NULL CHECK (role IN ('author','recipient')), -- author or recipient 

82 ordinal integer NOT NULL, -- position within the role as the source lists it 

83 display_name text, -- the display name as given 

84 address text, -- email address or handle 

85 normalized_name text, -- lower case, unaccented, whitespace collapsed 

86 normalized_address text, -- lower case, unaccented, whitespace collapsed 

87 provenance text NOT NULL CHECK (provenance IN ('source','connector')), -- source file or connector 

88 CHECK (display_name IS NOT NULL OR address IS NOT NULL), 

89 PRIMARY KEY (deployment_id, version_id, role, ordinal), 

90 FOREIGN KEY (deployment_id, version_id) REFERENCES document_metadata (deployment_id, version_id) ON DELETE CASCADE 

91); 

92COMMENT ON TABLE document_people IS 

93 'D134 the people in a version''s authors/recipients, one row each, for filtering by name or address. Hard forget deletes the lineage''s rows.'; 

94CREATE INDEX ix_document_people_name ON document_people USING gin (normalized_name gin_trgm_ops); 

95CREATE INDEX ix_document_people_address ON document_people (deployment_id, normalized_address); 

96"""

98_BACKFILL_METADATA = f""" 

99INSERT INTO document_metadata ( 

100 deployment_id, version_id, doc_id, family, file_name, source_path, title, 

101 created_at, modified_at, language, thread_ref, metadata_mapping_version 

102) 

103SELECT v.deployment_id, v.version_id, v.doc_id, 

104 CASE 

105 WHEN m.mime IN ('text/markdown', 'text/x-markdown') THEN 'markdown' 

106 WHEN m.mime IN ('text/html', 'application/xhtml+xml') THEN 'html' 

107 WHEN m.mime = 'application/pdf' THEN 'pdf' 

108 WHEN m.mime LIKE 'image/%' THEN 'image' 

109 WHEN m.mime LIKE 'audio/%' THEN 'audio' 

110 WHEN m.mime LIKE 'video/%' THEN 'video' 

111 WHEN m.mime IN ('application/msword', 'application/rtf') 

112 OR m.mime LIKE 'application/vnd.openxmlformats-officedocument.%' 

113 OR m.mime LIKE 'application/vnd.oasis.opendocument.%' 

114 OR m.mime LIKE 'application/vnd.ms-excel%' 

115 OR m.mime LIKE 'application/vnd.ms-powerpoint%' THEN 'office' 

116 WHEN m.mime LIKE 'text/%' THEN 'text' 

117 ELSE 'other' 

118 END, 

119 NULL, NULL, NULL, 

120 v.published_at, v.source_modified_at, v.language, v.thread_ref, 

121 '{BACKFILL_MAPPING_VERSION}' 

122FROM document_versions v 

123JOIN LATERAL ( 

124 SELECT lower(btrim(split_part(c.mime, ';', 1))) AS mime 

125 FROM content_objects c 

126 WHERE c.deployment_id = v.deployment_id AND c.content_hash = v.content_hash 

127) m ON true 

128WHERE NOT EXISTS ( 

129 SELECT 1 FROM forget_manifests f 

130 WHERE f.deployment_id = v.deployment_id AND f.doc_id = v.doc_id 

131) 

132ON CONFLICT (deployment_id, version_id) DO NOTHING 

133"""

135_BACKFILL_NAMES = f""" 

136INSERT INTO document_names ( 

137 deployment_id, version_id, observed_at, title, name_text, origin 

138) 

139SELECT md.deployment_id, md.version_id, v.ingested_at, d.title, btrim(d.title), 

140 'backfill' 

141FROM document_metadata md 

142JOIN document_versions v 

143 ON v.deployment_id = md.deployment_id AND v.version_id = md.version_id 

144JOIN documents d ON d.deployment_id = v.deployment_id AND d.doc_id = v.doc_id 

145WHERE md.metadata_mapping_version = '{BACKFILL_MAPPING_VERSION}' 

146 AND btrim(coalesce(d.title, '')) <> '' 

147ON CONFLICT (deployment_id, version_id, observed_at) DO NOTHING 

148"""
